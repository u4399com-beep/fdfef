    23→  'bg-emerald-100 text-emerald-700',
    24→  'bg-rose-100 text-rose-600',
    25→  'bg-amber-100 text-amber-700',

    30→function badgeClass(category: string): string {
    31→  let hash = 0
    32→  for (let i = 0; i < category.length; i += 1) {
    33→    hash = (hash * 31 + category.charCodeAt(i)) % 997
    34→  }
    35→  return BADGES[hash % BADGES.length]
    36→}

   467→/* ================= Chapter ================= */
   468→
   469→function MagazineChapter({ site, data, loading, onNavigate }: ThemeProps) {
   470→  const chapter = data.chapter
   471→  const paragraphs = chapter
   472→    ? chapter.content
   473→        .split('\n')
   474→        .map((p) => p.trim())
   475→        .filter(Boolean)
   476→    : []
   477→
   478→  return (
   479→    <div className="flex min-h-screen flex-col bg-stone-50 text-stone-900">
   480→      <Header site={site} onNavigate={onNavigate} />
   481→      <main className="mx-auto w-full max-w-3xl flex-1 px-4">
   482→        <Crumbs
   483→          trail={[
   484→            { label: '首页', onSelect: () => onNavigate({ type: 'home' }) },
   485→            chapter
   486→              ? {
   487→                  label: chapter.bookTitle,
   488→                  onSelect: () => onNavigate({ type: 'book', bookId: chapter.bookId }),
   489→                }
   490→              : { label: '书籍' },
   491→            { label: chapter?.title ?? '正文' },
   492→          ]}
   493→        />
   494→
   495→        {loading ? (
   496→          <ChapterSkeleton />
   497→        ) : !chapter ? (
   498→          <div className="py-24 text-center">
   499→            <p className="text-sm text-stone-500">未找到该章节内容。</p>
   500→            <button
   501→              type="button"
   502→              onClick={() => onNavigate({ type: 'home' })}
   503→              className="mt-6 min-h-[44px] rounded-full bg-stone-900 px-6 text-sm font-semibold text-white transition-colors hover:bg-stone-700"
   504→            >
   505→              返回首页
   506→            </button>
   507→          </div>
   508→        ) : (
   509→          <>
   510→            <article className="rounded-3xl bg-white p-6 shadow-sm ring-1 ring-stone-200/70 sm:p-10">
   511→              <h1 className="text-2xl font-black tracking-tight text-stone-900 sm:text-3xl">
   512→                {chapter.title}
   513→              </h1>
   514→              <p className="mt-2 flex flex-wrap items-center gap-x-3 gap-y-1 text-sm text-stone-400">
   515→                <span>《{chapter.bookTitle}》</span>
   516→                <span aria-hidden="true">·</span>
   517→                <span>第 {chapter.order} 章</span>
   518→              </p>
   519→              <div className="mt-6 border-t border-stone-100 pt-6" aria-hidden="true" />
   520→              <div className="space-y-5">
   521→                {paragraphs.map((p, i) => (
   522→                  <p key={i} className="text-[17px] leading-8 text-stone-700">
   523→                    {p}
   524→                  </p>
   525→                ))}
   526→              </div>
   527→            </article>
   528→            <nav
   529→              aria-label="章节导航"
   530→              className="mt-6 flex flex-col gap-3 pb-4 sm:flex-row sm:justify-between"
   531→            >
   532→              <button
   533→                type="button"
   534→                disabled={!chapter.prevId}
   535→                onClick={() => {
   536→                  if (chapter.prevId) onNavigate({ type: 'chapter', chapterId: chapter.prevId })
   537→                }}
   538→                className="inline-flex min-h-[44px] items-center justify-center gap-2 rounded-full border border-stone-200 bg-white px-6 text-sm font-semibold text-stone-600 shadow-sm transition-colors hover:bg-stone-100 disabled:cursor-not-allowed disabled:opacity-40"
   539→              >
   540→                <ArrowLeft className="h-4 w-4" aria-hidden="true" />

   542→              </button>
   543→              <button
   544→                type="button"
   545→                onClick={() => onNavigate({ type: 'toc', bookId: chapter.bookId })}

   550→              <button
   551→                type="button"
   552→                disabled={!chapter.nextId}
   553→                onClick={() => {
   554→                  if (chapter.nextId) onNavigate({ type: 'chapter', chapterId: chapter.nextId })
   555→                }}

   560→              </button>
   561→            </nav>

   563→        )}
   564→      </main>
   565→      <Footer site={site} />
   566→    </div>

   569→
   570→/* ================= Toc（完整章节目录页） ================= */
   571→
   572→const TOC_PAGE_SIZE = 100
   573→
   574→function MagazineToc({ site, data, loading, onNavigate }: ThemeProps) {
   575→  const [page, setPage] = useState(1)
   576→  const book = data.book
   577→  const chapters = book?.chapters ?? []
   578→  const totalPages = Math.max(1, Math.ceil(chapters.length / TOC_PAGE_SIZE))
   579→  const safePage = Math.min(page, totalPages)
   580→  const visible = chapters.slice((safePage - 1) * TOC_PAGE_SIZE, safePage * TOC_PAGE_SIZE)
   581→
   582→  return (
   583→    <div className="flex min-h-screen flex-col bg-stone-50 text-stone-900">
   584→      <Header site={site} onNavigate={onNavigate} />
   585→      <main className="mx-auto w-full max-w-5xl flex-1 px-4">
   586→        <Crumbs
   587→          trail={[
   588→            { label: '首页', onSelect: () => onNavigate({ type: 'home' }) },
   589→            {
   590→              label: book?.title ?? '书籍',
   591→              onSelect: book ? () => onNavigate({ type: 'book', bookId: book.id }) : undefined,
   592→            },
   593→            { label: '章节目录' },
   594→          ]}
   595→        />
   596→
   597→        {loading ? (
   598→          <BookPageSkeleton />
   599→        ) : !book ? (
   600→          <div className="py-24 text-center">
   601→            <BookOpen className="mx-auto h-10 w-10 text-stone-300" aria-hidden="true" />
   602→            <p className="mt-4 text-sm text-stone-500">未找到该书籍，可能已被移除。</p>
   603→            <button
   604→              type="button"
   605→              onClick={() => onNavigate({ type: 'home' })}
   606→              className="mt-6 min-h-[44px] rounded-full bg-stone-900 px-6 text-sm font-semibold text-white transition-colors hover:bg-stone-700"
   607→            >
   608→              返回首页
   609→            </button>
   610→          </div>
   611→        ) : (
   612→          <>
   613→            <div className="flex flex-wrap items-end justify-between gap-3">
   614→              <h1 className="text-2xl font-black tracking-tight text-stone-900 sm:text-3xl">
   615→                {book.title}
   616→                <span className="ml-3 text-base font-medium text-stone-400">章节目录</span>
   617→              </h1>
   618→              <p className="text-sm text-stone-500">
   619→                {book.author} · 共 {book.totalChapters} 章
   620→              </p>
   621→            </div>
   622→
   623→            <nav
   624→              aria-label="目录分页"
   625→              className="mt-4 flex flex-wrap items-center gap-2 border-y border-stone-200 py-3"
   626→            >
   627→              <button
   628→                type="button"
   629→                disabled={safePage <= 1}
   630→                onClick={() => setPage((p) => Math.max(1, p - 1))}
   631→                className="min-h-[36px] rounded-full border border-stone-300 bg-white px-4 text-sm font-semibold text-stone-600 transition-colors hover:bg-stone-100 disabled:cursor-not-allowed disabled:opacity-40"
   632→              >
   633→                上一页
   634→              </button>
   635→              <span className="text-sm text-stone-500">
   636→                第 {safePage} / {totalPages} 页
   637→              </span>
   638→              <button
   639→                type="button"
   640→                disabled={safePage >= totalPages}
   641→                onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   642→                className="min-h-[36px] rounded-full border border-stone-300 bg-white px-4 text-sm font-semibold text-stone-600 transition-colors hover:bg-stone-100 disabled:cursor-not-allowed disabled:opacity-40"
   643→              >
   644→                下一页
   645→              </button>
   646→            </nav>
   647→
   648→            {visible.length === 0 ? (
   649→              <p className="py-10 text-center text-sm text-stone-400">暂无章节。</p>
   650→            ) : (
   651→              <div className="mb-2 mt-4 rounded-3xl bg-white p-4 shadow-sm ring-1 ring-stone-200/70 sm:p-6">
   652→                <ul className="grid grid-cols-1 gap-1 md:grid-cols-2">
   653→                  {visible.map((ch) => (
   654→                    <ChapterRow key={ch.id} chapter={ch} onNavigate={onNavigate} />
   655→                  ))}
   656→                </ul>
   657→              </div>
   658→            )}
   659→          </>
   660→        )}
   661→      </main>
   662→      <Footer site={site} />
   663→    </div>
   664→  )
   665→}

   667→/* ================= Keyword ================= */
   668→
   669→function MagazineKeyword({
   670→  site,
   671→  data,
   672→  loading,
   673→  onNavigate,
   674→  keyword,
   675→}: ThemeProps & { keyword: string }) {
   676→  const books = data.keywordBooks ?? []
   677→  const goMain = () =>
   678→    onNavigate(site.mainBookId ? { type: 'book', bookId: site.mainBookId } : { type: 'home' })
   679→  const related = Array.from(
   680→    new Set(
   681→      books
   682→        .flatMap((b) => [...b.keywords, ...b.suggestKeywords])
   683→        .filter((k) => k !== keyword),
   684→    ),
   685→  ).slice(0, 12)
   686→
   687→  return (
   688→    <div className="flex min-h-screen flex-col bg-stone-50 text-stone-900">
   689→      <Header site={site} onNavigate={onNavigate} />
   690→      <main className="mx-auto w-full max-w-6xl flex-1 px-4">
   691→        <Crumbs
   692→          trail={[
   693→            { label: '首页', onSelect: () => onNavigate({ type: 'home' }) },
   694→            { label: `关键词：${keyword}` },
   695→          ]}
   696→        />
   697→
   698→        {/* 最醒目的主书籍入口 */}
   699→        <button
   700→          type="button"
   701→          onClick={goMain}
   702→          className="group flex min-h-[44px] w-full items-center justify-between gap-4 rounded-2xl bg-gradient-to-r from-emerald-500 via-teal-500 to-emerald-600 px-6 py-5 text-left shadow-lg shadow-emerald-500/20 transition-all hover:-translate-y-0.5 hover:shadow-xl"
   703→        >
   704→          <span className="flex items-center gap-3 text-base font-extrabold text-white">
   705→            <Sparkles className="h-5 w-5" aria-hidden="true" />
   706→            进入主书籍信息页
   707→          </span>
   708→          <ArrowRight
   709→            className="h-5 w-5 text-white transition-transform group-hover:translate-x-1"
   710→            aria-hidden="true"
   711→          />
   712→        </button>
   713→
   714→        <section className="py-10">
   715→          <h1 className="text-3xl font-black tracking-tight text-stone-900 sm:text-4xl">
   716→            {keyword}
   717→          </h1>
   718→          <p className="mt-3 text-sm text-stone-500">与《{keyword}》相关的书籍与内容</p>
   719→        </section>
   720→
   721→        {loading ? (
   722→          <GridSkeleton count={4} />
   723→        ) : books.length === 0 ? (
   724→          <div className="py-20 text-center">
   725→            <BookOpen className="mx-auto h-10 w-10 text-stone-300" aria-hidden="true" />
   726→            <p className="mt-4 text-sm text-stone-500">暂无与「{keyword}」相关的书籍。</p>
   727→          </div>
   728→        ) : (
   729→          <section
   730→            aria-label="相关书籍"
   731→            className="grid grid-cols-1 gap-5 pb-4 sm:grid-cols-2 md:grid-cols-3 lg:grid-cols-4"
   732→          >
   733→            {books.map((book) => (
   734→              <BookCard key={book.id} book={book} onNavigate={onNavigate} />
   735→            ))}
   736→          </section>
   737→        )}
   738→
   739→        {related.length > 0 && (
   740→          <section aria-label="相关关键词" className="mt-6 pb-4">
   741→            <h2 className="text-sm font-bold text-stone-400">相关关键词</h2>
   742→            <div className="mt-3 flex flex-wrap gap-2">
   743→              {related.map((k) => (
   744→                <button
   745→                  key={k}
   746→                  type="button"
   747→                  onClick={() => onNavigate({ type: 'keyword', keyword: k })}
   748→                  className="min-h-[44px] rounded-full bg-white px-4 text-sm font-medium text-stone-600 ring-1 ring-stone-200 transition-colors hover:bg-emerald-50 hover:text-emerald-700"
   749→                >
   750→                  # {k}
   751→                </button>
   752→              ))}
   753→            </div>
   754→          </section>
   755→        )}
   756→      </main>
   757→      <Footer site={site} />
   758→    </div>
   759→  )
   760→}

   762→/* ================= 主组件 ================= */
   763→
   764→export function ThemeMagazine(props: ThemeProps): ReactElement {
   765→  const { view } = props
   766→  switch (view.type) {
   767→    case 'home':
   768→      return <MagazineHome {...props} />
   769→    case 'book':
   770→      return <MagazineBook {...props} />
   771→    case 'toc':
   772→      return <MagazineToc {...props} />
   773→    case 'chapter':
   774→      return <MagazineChapter {...props} />
   775→    case 'keyword':
   776→      return <MagazineKeyword {...props} keyword={view.keyword} />
   777→    default:
   778→      return <MagazineHome {...props} />
   779→  }
   780→}
