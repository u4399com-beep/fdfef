   441→/* ================= Toc（完整章节目录页） ================= */
   442→
   443→const TOC_PAGE_SIZE = 100
   444→
   445→function ClassicToc({ site, data, loading, onNavigate }: ThemeProps) {
   446→  const [page, setPage] = useState(1)
   447→  const book = data.book
   448→  const chapters = book?.chapters ?? []
   449→  const totalPages = Math.max(1, Math.ceil(chapters.length / TOC_PAGE_SIZE))
   450→  const safePage = Math.min(page, totalPages)
   451→  const visible = chapters.slice((safePage - 1) * TOC_PAGE_SIZE, safePage * TOC_PAGE_SIZE)
   452→
   453→  return (
   454→    <div className="flex min-h-screen flex-col bg-amber-50 text-stone-800">
   455→      <Header site={site} onNavigate={onNavigate} />
   456→      <main className="mx-auto w-full max-w-5xl flex-1 px-4">
   457→        <Breadcrumb
   458→          items={[
   459→            { label: '首页', onSelect: () => onNavigate({ type: 'home' }) },
   460→            {
   461→              label: book?.title ?? '书籍',
   462→              onSelect: book ? () => onNavigate({ type: 'book', bookId: book.id }) : undefined,
   463→            },
   464→            { label: '章节目录' },
   465→          ]}
   466→        />
   467→
   468→        {loading ? (
   469→          <BookSkeleton />
   470→        ) : !book ? (
   471→          <div className="py-24 text-center">
   472→            <p className="font-serif text-lg text-stone-500">未找到该书籍，可能已被移除。</p>
   473→            <button
   474→              type="button"
   475→              onClick={() => onNavigate({ type: 'home' })}
   476→              className="mt-6 min-h-[44px] rounded-sm bg-amber-800 px-6 font-serif text-sm text-amber-50 transition-colors hover:bg-amber-900"
   477→            >
   478→              返回首页
   479→            </button>
   480→          </div>
   481→        ) : (
   482→          <>
   483→            <div className="flex flex-wrap items-end justify-between gap-3">
   484→              <h1 className="font-serif text-2xl font-bold text-stone-900">
   485→                {book.title}
   486→                <span className="ml-3 text-base font-normal text-stone-400">章节目录</span>
   487→              </h1>
   488→              <p className="font-serif text-sm text-stone-500">
   489→                {book.author} · 共 {book.totalChapters} 章
   490→              </p>
   491→            </div>
   492→
   493→            <nav
   494→              aria-label="目录分页"
   495→              className="mt-4 flex flex-wrap items-center gap-2 border-y border-amber-200 py-3"
   496→            >
   497→              <button
   498→                type="button"
   499→                disabled={safePage <= 1}
   500→                onClick={() => setPage((p) => Math.max(1, p - 1))}
   501→                className="min-h-[36px] rounded-sm border border-amber-300 bg-amber-50 px-3 font-serif text-sm text-amber-900 transition-colors hover:bg-amber-100 disabled:cursor-not-allowed disabled:opacity-40"
   502→              >
   503→                上一页
   504→              </button>
   505→              <span className="font-serif text-sm text-stone-500">
   506→                第 {safePage} / {totalPages} 页
   507→              </span>
   508→              <button
   509→                type="button"
   510→                disabled={safePage >= totalPages}
   511→                onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   512→                className="min-h-[36px] rounded-sm border border-amber-300 bg-amber-50 px-3 font-serif text-sm text-amber-900 transition-colors hover:bg-amber-100 disabled:cursor-not-allowed disabled:opacity-40"
   513→              >
   514→                下一页
   515→              </button>
   516→            </nav>
   517→
   518→            {visible.length === 0 ? (
   519→              <p className="py-10 text-center font-serif text-stone-400">暂无章节。</p>
   520→            ) : (
   521→              <ul className="grid grid-cols-1 border-x border-b border-amber-200 md:grid-cols-2">
   522→                {visible.map((ch) => (
   523→                  <ChapterLine key={ch.id} chapter={ch} onNavigate={onNavigate} />
   524→                ))}
   525→              </ul>
   526→            )}
   527→          </>
   528→        )}
   529→      </main>
   530→      <Footer site={site} />
   531→    </div>
   532→  )
   533→}

   535→/* ================= Chapter ================= */
   536→
   537→function ClassicChapter({ site, data, loading, onNavigate }: ThemeProps) {
   538→  const chapter = data.chapter
   539→  const paragraphs = chapter
   540→    ? chapter.content
   541→        .split('\n')
   542→        .map((p) => p.trim())
   543→        .filter(Boolean)
   544→    : []
   545→
   546→  return (
   547→    <div className="flex min-h-screen flex-col bg-amber-50 text-stone-800">
   548→      <Header site={site} onNavigate={onNavigate} />
   549→      <main className="mx-auto w-full max-w-3xl flex-1 px-4">
   550→        <Breadcrumb
   551→          items={[
   552→            { label: '首页', onSelect: () => onNavigate({ type: 'home' }) },
   553→            {
   554→              label: chapter?.bookTitle ?? '书籍',
   555→              onSelect: chapter
   556→                ? () => onNavigate({ type: 'book', bookId: chapter.bookId })
   557→                : undefined,
   558→            },
   559→            { label: chapter?.title ?? '正文' },
   560→          ]}
   561→        />
   562→
   563→        {loading ? (
   564→          <ChapterSkeleton />
   565→        ) : !chapter ? (
   566→          <div className="py-24 text-center">
   567→            <p className="font-serif text-lg text-stone-500">未找到该章节内容。</p>
   568→            <button
   569→              type="button"
   570→              onClick={() => onNavigate({ type: 'home' })}
   571→              className="mt-6 min-h-[44px] rounded-sm bg-amber-800 px-6 font-serif text-sm text-amber-50 transition-colors hover:bg-amber-900"
   572→            >
   573→              返回首页
   574→            </button>
   575→          </div>
   576→        ) : (
   577→          <article className="pb-6">
   578→            <h1 className="text-center font-serif text-2xl font-bold text-stone-900 sm:text-3xl">
   579→              {chapter.title}
   580→            </h1>
   581→            <p className="mt-3 text-center font-serif text-sm text-stone-500">
   582→              《{chapter.bookTitle}》· 第 {chapter.order} 章
   583→            </p>
   584→            <div className="mb-2 mx-auto mt-8 flex items-center gap-3" aria-hidden="true">
   585→              <span className="h-px flex-1 bg-amber-200" />
   586→              <BookOpen className="h-4 w-4 text-amber-600" />
   587→              <span className="h-px flex-1 bg-amber-200" />
   588→            </div>
   589→            <div className="space-y-6 py-4 font-serif text-lg leading-loose text-stone-800">
   590→              {paragraphs.map((p, i) => (
   591→                <p key={i}>{p}</p>
   592→              ))}
   593→            </div>
   594→            <nav
   595→              aria-label="章节导航"
   596→              className="mt-8 flex flex-col items-stretch justify-center gap-3 border-t border-amber-200 pt-6 sm:flex-row"
   597→            >
   598→              <button
   599→                type="button"
   600→                disabled={!chapter.prevId}
   601→                onClick={() => {
   602→                  if (chapter.prevId) onNavigate({ type: 'chapter', chapterId: chapter.prevId })
   603→                }}
   604→                className="flex min-h-[44px] items-center justify-center gap-2 rounded-sm border border-amber-300 bg-amber-50 px-6 font-serif text-sm text-amber-900 transition-colors hover:bg-amber-100 disabled:cursor-not-allowed disabled:opacity-40"
   605→              >
   606→                <ArrowLeft className="h-4 w-4" aria-hidden="true" />
   607→                上一章
   608→              </button>
   609→              <button
   610→                type="button"
   611→                onClick={() => onNavigate({ type: 'toc', bookId: chapter.bookId })}
   612→                className="flex min-h-[44px] items-center justify-center gap-2 rounded-sm bg-amber-800 px-6 font-serif text-sm text-amber-50 transition-colors hover:bg-amber-900"
   613→              >
   614→                <List className="h-4 w-4" aria-hidden="true" />
   615→                返回目录
   616→              </button>
   617→              <button
   618→                type="button"
   619→                disabled={!chapter.nextId}
   620→                onClick={() => {
   621→                  if (chapter.nextId) onNavigate({ type: 'chapter', chapterId: chapter.nextId })
   622→                }}
   623→                className="flex min-h-[44px] items-center justify-center gap-2 rounded-sm border border-amber-300 bg-amber-50 px-6 font-serif text-sm text-amber-900 transition-colors hover:bg-amber-100 disabled:cursor-not-allowed disabled:opacity-40"
   624→              >
   625→                下一章
   626→                <ArrowRight className="h-4 w-4" aria-hidden="true" />
   627→              </button>
   628→            </nav>
   629→          </article>
   630→        )}
   631→      </main>
   632→      <Footer site={site} />
   633→    </div>
   634→  )
   635→}

   637→/* ================= Keyword ================= */
   638→
   639→function ClassicKeyword({
   640→  site,
   641→  data,
   642→  loading,
   643→  onNavigate,
   644→  keyword,
   645→}: ThemeProps & { keyword: string }) {
   646→  const books = data.keywordBooks ?? []
   647→  const goMain = () =>
   648→    onNavigate(site.mainBookId ? { type: 'book', bookId: site.mainBookId } : { type: 'home' })
   649→  const relatedKeywords = Array.from(
   650→    new Set(
   651→      books
   652→        .flatMap((b) => [...b.keywords, ...b.suggestKeywords])
   653→        .filter((k) => k !== keyword),
   654→    ),
   655→  ).slice(0, 12)
   656→
   657→  return (
   658→    <div className="flex min-h-screen flex-col bg-amber-50 text-stone-800">
   659→      <Header site={site} onNavigate={onNavigate} />
   660→      <main className="mx-auto w-full max-w-5xl flex-1 px-4">
   661→        <Breadcrumb
   662→          items={[
   663→            { label: '首页', onSelect: () => onNavigate({ type: 'home' }) },
   664→            { label: `关键词：${keyword}` },
   665→          ]}
   666→        />
   667→
   668→        {/* 最醒目的主书籍入口 */}
   669→        <button
   670→          type="button"
   671→          onClick={goMain}
   672→          className="flex min-h-[44px] w-full items-center justify-center gap-3 rounded-sm border-2 border-amber-800 bg-amber-800 px-6 py-4 font-serif text-base font-bold text-amber-50 shadow-sm transition-colors hover:bg-amber-900"
   673→        >
   674→          <BookOpen className="h-5 w-5" aria-hidden="true" />
   675→          进入主书籍信息页
   676→          <ArrowRight className="h-4 w-4" aria-hidden="true" />
   677→        </button>
   678→
   679→        <section className="py-10 text-center">
   680→          <h1 className="font-serif text-3xl font-bold text-stone-900 sm:text-4xl">{keyword}</h1>
   681→          <p className="mt-3 font-serif text-stone-500">与《{keyword}》相关的书籍与内容</p>
   682→        </section>
   683→
   684→        {loading ? (
   685→          <ListSkeleton rows={3} />
   686→        ) : books.length === 0 ? (
   687→          <p className="py-16 text-center font-serif text-stone-500">
   688→            暂无与「{keyword}」相关的书籍。
   689→          </p>
   690→        ) : (
   691→          <ul className="divide-y divide-amber-100">
   692→            {books.map((book) => (
   693→              <BookRow key={book.id} book={book} onNavigate={onNavigate} />
   694→            ))}
   695→          </ul>
   696→        )}
   697→
   698→        {relatedKeywords.length > 0 && (
   699→          <section aria-label="相关关键词" className="mt-10 border-t border-amber-200 py-6">
   700→            <h2 className="font-serif text-sm text-stone-500">相关关键词</h2>
   701→            <div className="mt-3 flex flex-wrap gap-2">
   702→              {relatedKeywords.map((k) => (
   703→                <button
   704→                  key={k}
   705→                  type="button"
   706→                  onClick={() => onNavigate({ type: 'keyword', keyword: k })}
   707→                  className="min-h-[44px] rounded-sm bg-amber-100/80 px-3 py-1 font-serif text-sm text-amber-900 transition-colors hover:bg-amber-200"
   708→                >
   709→                  {k}
   710→                </button>
   711→              ))}
   712→            </div>
   713→          </section>
   714→        )}
   715→      </main>
   716→      <Footer site={site} />
   717→    </div>
   718→  )
   719→}

   721→/* ================= 主组件 ================= */
   722→
   723→export function ThemeClassic(props: ThemeProps): ReactElement {
   724→  const { view } = props
   725→  switch (view.type) {
   726→    case 'home':
   727→      return <ClassicHome {...props} />
   728→    case 'book':
   729→      return <ClassicBook {...props} />
   730→    case 'toc':
   731→      return <ClassicToc {...props} />
   732→    case 'chapter':
   733→      return <ClassicChapter {...props} />
   734→    case 'keyword':
   735→      return <ClassicKeyword {...props} keyword={view.keyword} />
   736→    default:
   737→      return <ClassicHome {...props} />
   738→  }
   739→}
   740→
   741→export default ThemeClassic
