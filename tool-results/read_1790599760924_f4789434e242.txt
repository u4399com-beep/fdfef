    39→export function randomInt(min: number, max: number): number {
    40→  if (!Number.isFinite(min) || !Number.isFinite(max)) return min
    41→  const lo = Math.max(0, Math.ceil(min))
    42→  const hi = Math.max(lo, Math.floor(max))
    43→  if (lo === hi) return lo
    44→  return lo + Math.floor(Math.random() * (hi - lo + 1))
    45→}

    65→interface JarCookie {
    66→  name: string
    67→  value: string
    68→  domain?: string
    69→  path?: string
    70→  expires?: number
    71→}
    72→
    73→class CookieJar {
    74→  private store: Map<string, JarCookie[]>
    75→
    76→  constructor(store?: Map<string, JarCookie[]>) {
    77→    this.store = store ?? new Map()
    78→  }
    79→
    80→  /** 解析 Set-Cookie 响应头并入 jar */
    81→  absorbFromFetch(host: string, headers: Headers): void {
    82→    const raw = typeof headers.getSetCookie === 'function' ? headers.getSetCookie() : []
    83→    if (!raw.length) return
    84→    const list = this.store.get(host) ?? []
    85→    for (const line of raw) {
    86→      const [pair, ...attrs] = line.split(';')
    87→      const idx = pair.indexOf('=')
    88→      if (idx <= 0) continue
    89→      const name = pair.slice(0, idx).trim()
    90→      const value = pair.slice(idx + 1).trim()
    91→      const lower = attrs.map((a) => a.trim().toLowerCase())
    92→      const expired = lower.some((a) => a.startsWith('expires=') && new Date(a.slice(8)).getTime() < Date.now())
    93→      const pos = list.findIndex((c) => c.name === name)
    94→      if (expired) {
    95→        if (pos >= 0) list.splice(pos, 1)
    96→        continue
    97→      }
    98→      if (pos >= 0) list.splice(pos, 1)
    99→      list.push({ name, value })
   100→    }
   101→    this.store.set(host, list)
   102→  }
   103→
   104→  /** Playwright context.cookies() 结果并入 jar */
   105→  absorbFromBrowser(originHost: string, cookies: { name: string; value: string; expires?: number }[]): void {
   106→    const list = this.store.get(originHost) ?? []
   107→    for (const c of cookies) {
   108→      if (!c.name) continue
   109→      const pos = list.findIndex((x) => x.name === c.name)
   110→      if (pos >= 0) list.splice(pos, 1)
   111→      if (c.expires && c.expires > 0 && c.expires * 1000 < Date.now()) continue
   112→      list.push({ name: c.name, value: c.value, expires: c.expires })
   113→    }
   114→    this.store.set(originHost, list)
   115→  }
   116→
   117→  /** 取指定 URL 的 Cookie 请求头 */
   118→  header(url: string): string {
   119→    let host = ''
   120→    try {
   121→      host = new URL(url).hostname
   122→    } catch {
   123→      return ''
   124→    }
   125→    // 允许父子域共享（.kelexs.com 场景）：精确 host + 以点开头的域后缀
   126→    const list: JarCookie[] = []
   127→    for (const [key, cookies] of this.store) {
   128→      if (key === host || host.endsWith(key.replace(/^\./, ''))) list.push(...cookies)
   129→    }
   130→    if (!list.length) return ''
   131→    return list.map((c) => `${c.name}=${c.value}`).join('; ')
   132→  }
   133→
   134→  count(): number {
   135→    let n = 0
   136→    for (const cookies of this.store.values()) n += cookies.length
   137→    return n
   138→  }
   139→}

   152→// 全局同域节流器：无论任务线程数如何，同一域名的请求间隔不小于最小值，
   153→// 从源头避免高频请求触发目标站 WAF（反反爬核心基础设施）
   154→// ============================================================
   155→
   156→// 状态数据挂 globalThis（跨热重载保留），类实例每次模块加载新建（类代码始终最新）——
   157→// 从根上避免 dev 热重载后旧实例缺少新方法的问题
   158→interface ThrottleState {
   159→  queues: Map<string, Promise<void>>
   160→  blockedUntil: Map<string, number>
   161→}
   162→const THROTTLE_STATE_KEY = '__novelThrottleState'
   163→const globalForThrottle = globalThis as unknown as { [THROTTLE_STATE_KEY]?: ThrottleState }
   164→const throttleState: ThrottleState =
   165→  globalForThrottle[THROTTLE_STATE_KEY] ?? { queues: new Map(), blockedUntil: new Map() }
   166→globalForThrottle[THROTTLE_STATE_KEY] = throttleState
   167→
   168→class DomainThrottle {
   169→  private queues = throttleState.queues
   170→  private blockedUntil = throttleState.blockedUntil
   171→
   172→  /**
   173→   * 同域串行化：每个请求至少间隔 minGapMs。
   174→   * 若该域处于封禁冷却期（reportBlock），自动等待冷却结束后再入队。
   175→   */
   176→  async wait(url: string, customGap?: number): Promise<void> {
   177→    let host = ''
   178→    try {
   179→      host = new URL(url).hostname
   180→    } catch {
   181→      return
   182→    }
   183→    // 封禁冷却期：等待解除后继续（避免在封禁期继续请求加剧封禁）
   184→    const blockedUntil = this.blockedUntil.get(host) ?? 0
   185→    const now = Date.now()
   186→    if (now < blockedUntil) {
   187→      await sleep(blockedUntil - now + 200)
   188→      this.blockedUntil.delete(host)
   189→    }
   190→    const gap = Math.max(0, customGap ?? 1200)
   191→    if (gap <= 0) return
   192→    const prev = this.queues.get(host) ?? Promise.resolve()
   193→    const next = prev.then(
   194→      () => new Promise<void>((r) => setTimeout(r, gap)),
   195→      () => new Promise<void>((r) => setTimeout(r, gap))
   196→    )
   197→    this.queues.set(host, next)
   198→    await next
   199→  }
   200→
   201→  /** 目标站封禁（硬 403）：设置同域全局冷却期 */
   202→  reportBlock(url: string, ms = 180_000): void {
   203→    try {
   204→      const host = new URL(url).hostname
   205→      const until = Date.now() + ms + Math.random() * 60_000
   206→      const prev = this.blockedUntil.get(host) ?? 0
   207→      if (until > prev) this.blockedUntil.set(host, until)
   208→    } catch {
   209→      /* ignore */
   210→    }
   211→  }
   212→}

   231→/** 判断 HTML 是否为 WAF/验证码挑战页或硬拒绝页（403 黑名单） */
   232→export function isWafChallengeHtml(html: string): boolean {
   233→  if (isHardDeniedHtml(html)) return true
   234→  if (!html || html.length > 20000) {
   235→    // 验证页很小；超长页面再做签名兜底检查（避免正文误判）
   236→    return WAF_SIGNATURES.slice(0, 2).some((re) => re.test(html))
   237→  }
   238→  return WAF_SIGNATURES.some((re) => re.test(html))
   239→}

   241→/** 硬 403/封禁页（IP 黑名单，需向调用方明确报告而非静默解析为空） */
   242→export function isHardDeniedHtml(html: string): boolean {
   243→  if (!html || html.length > 4000) return false
   244→  return /<title>\s*403\s*Forbidden\s*<\/title>/i.test(html) || /\b403 Forbidden\b/.test(html)
   245→}

   251→function detectCharset(buffer: Buffer, contentType: string, fallback: string): string {
   252→  const explicit = fallback !== 'auto' ? fallback : ''
   253→  if (explicit) return explicit
   254→  const ctMatch = /charset=["']?([\w-]+)/i.exec(contentType)
   255→  if (ctMatch) return ctMatch[1].toLowerCase()
   256→  const head = buffer.subarray(0, 4096).toString('latin1')
   257→  const meta = /<meta[^>]+charset=["']?([\w-]+)/i.exec(head) ?? /charset=["']([\w-]+)/i.exec(head)
   258→  if (meta) return meta[1].toLowerCase()
   259→  return 'utf-8'
   260→}
   261→
   262→function decodeBuffer(buffer: Buffer, charset: string): string {
   263→  const cs = charset.toLowerCase()
   264→  if (cs === 'utf-8' || cs === 'utf8' || cs === 'ascii') {
   265→    return new TextDecoder('utf-8').decode(buffer)
   266→  }
   267→  // 动态加载 iconv-lite 以处理 gbk/gb2312/big5
   268→  try {
   269→    // eslint-disable-next-line @typescript-eslint/no-require-imports
   270→    const iconv = require('iconv-lite') as {
   271→      decode: (b: Buffer, enc: string) => string
   272→      encodingExists: (enc: string) => boolean
   273→    }
   274→    if (iconv.encodingExists(cs === 'gb2312' ? 'gbk' : cs)) {
   275→      return iconv.decode(buffer, cs === 'gb2312' ? 'gbk' : cs)
   276→    }
   277→  } catch {
   278→    /* fallthrough */
   279→  }
   280→  return new TextDecoder('utf-8').decode(buffer)
   281→}

   287→/** 合并两组 cookie 串：override 中同名键优先，base 中独有键保留（jar 通行 cookie 与显式 cookie 合并） */
   288→function mergeCookieStrings(base: string, override: string): string {
   289→  const parse = (s: string): [string, string][] =>
   290→    s
   291→      .split(';')
   292→      .map((c) => c.trim())
   293→      .filter(Boolean)
   294→      .map((c) => {
   295→        const idx = c.indexOf('=')
   296→        return idx <= 0 ? null : ([c.slice(0, idx).trim(), c.slice(idx + 1).trim()] as [string, string])
   297→      })
   298→      .filter((p): p is [string, string] => p !== null)
   299→  const map = new Map(parse(base))
   300→  for (const [k, v] of parse(override)) map.set(k, v)
   301→  return [...map.entries()].map(([k, v]) => `${k}=${v}`).join('; ')
   302→}

   304→function buildHeaders(cfg: FetchConfig, url: string, uaOverride?: string): Record<string, string> {
   305→  const ua = uaOverride ?? (cfg.rotateUA === false ? FIXED_UA : randomUA())
   306→  const headers: Record<string, string> = {
   307→    'User-Agent': ua,
   308→    Accept: 'text/html,application/xhtml+xml,application/xml;q=0.9,image/avif,image/webp,*/*;q=0.8',
   309→    'Accept-Language': 'zh-CN,zh;q=0.9,en;q=0.6',
   310→    'Cache-Control': 'no-cache',
   311→    Pragma: 'no-cache',
   312→    'Upgrade-Insecure-Requests': '1',
   313→  }
   314→  // Chrome 系 UA 注入配套 Client Hints（真实浏览器指纹）
   315→  const chromeVer = chromeMajorVersion(ua)
   316→  if (chromeVer) {
   317→    headers['sec-ch-ua'] = `"Chromium";v="${chromeVer}", "Google Chrome";v="${chromeVer}", "Not-A.Brand";v="99"`
   318→    headers['sec-ch-ua-mobile'] = /Mobile|Android|iPhone/.test(ua) ? '?1' : '?0'
   319→    headers['sec-ch-ua-platform'] = /Windows/i.test(ua)
   320→      ? '"Windows"'
   321→      : /Macintosh/i.test(ua)
   322→        ? '"macOS"'
   323→        : /Android/i.test(ua)
   324→          ? '"Android"'
   325→          : /iPhone|iPad/i.test(ua)
   326→            ? '"iOS"'
   327→            : '"Linux"'
   328→  }
   329→  if (cfg.referer) headers.Referer = cfg.referer
   330→  else {
   331→    // 默认 Referer 指纹：真实浏览器站内跳转必带同源 Referer，裸无 Referer 是明显爬虫特征
   332→    try {
   333→      headers.Referer = `${new URL(url).origin}/`
   334→    } catch {
   335→      /* 非法 URL 已在入口拦 */
   336→    }
   337→  }
   338→  // 显式 cookies 与 jar 会话 cookie 合并（显式优先同名键）：
   339→  // WAF 解题后新入 jar 的通行 cookie 不再被旧显式 cookie 完全屏蔽
   340→  if (cfg.cookies) {
   341→    const jarCookie = cookieJar.header(url)
   342→    headers.Cookie = jarCookie ? mergeCookieStrings(jarCookie, cfg.cookies) : cfg.cookies
   343→  } else {
   344→    const jarCookie = cookieJar.header(url)
   345→    if (jarCookie) headers.Cookie = jarCookie
   346→  }
   347→  if (cfg.headers) {
   348→    // 兼容 UI 保存的 JSON 字符串形式（对象按原样合并，字符串则解析后再合并）
   349→    let extra = cfg.headers as unknown
   350→    if (typeof extra === 'string') {
   351→      try {
   352→        extra = JSON.parse(extra)
   353→      } catch {
   354→        extra = null
   355→      }
   356→    }
   357→    if (extra && typeof extra === 'object') Object.assign(headers, extra)
   358→  }
   359→  return headers
   360→}

   362→/** 主入口：按策略抓取页面 */
   363→export async function fetchPage(url: string, cfg: FetchConfig = {}): Promise<FetchResult> {
   364→  if (!/^https?:\/\//i.test(url)) throw new Error(`非法 URL：${url}`)
   365→  const strategy = cfg.strategy ?? 'http'
   366→  const timeout = cfg.timeout ?? 20000
   367→  const started = Date.now()
   368→
   369→  if (strategy === 'playwright') return fetchWithPlaywright(url, cfg, timeout, started)
   370→  if (strategy === 'hyperbrowser') return fetchWithHyperbrowser(url, cfg, timeout, started)
   371→
   372→  // HTTP 直连策略（遇 WAF 自动升级浏览器渲染；同域全局节流）
   373→  await domainThrottle.wait(url, cfg.throttleGap)
   374→  let html = ''
   375→  let status = 0
   376→  let finalUrl = url
   377→  for (let attempt = 0; attempt < 2; attempt++) {
   378→    const res = await fetchWithRetry(url, buildHeaders(cfg, url), timeout, 2, cfg.rotateUA === false)
   379→    const buffer = Buffer.from(await res.arrayBuffer())
   380→    cookieJar.absorbFromFetch(new URL(res.url || url).hostname, res.headers)
   381→    const charset = detectCharset(buffer, res.headers.get('content-type') ?? '', cfg.encoding ?? 'auto')
   382→    html = decodeBuffer(buffer, charset)
   383→    status = res.status
   384→    finalUrl = res.url || url
   385→    if (isHardDeniedHtml(html)) {
   386→      // IP 已被目标站拉黑（硬 403）：全局冷却后向调用方明确报告
   387→      domainThrottle.reportBlock(finalUrl, 180_000)
   388→      throw new Error(`目标站拒绝访问（IP 临时封禁，HTTP 403）：${finalUrl}。已自动冷却 3 分钟`)
   389→    }
   390→    if (!isWafChallengeHtml(html)) {
   391→      return { html, status, finalUrl, strategy: 'http', elapsedMs: Date.now() - started }
   392→    }
   393→    // WAF 拦截：升级为 Playwright（带验证码求解），仅尝试一次
   394→    try {
   395→      const pwResult = await fetchWithPlaywright(url, cfg, timeout, started)
   396→      return { ...pwResult, elapsedMs: Date.now() - started }
   397→    } catch (e) {
   398→      if (attempt === 1) {
   399→        throw new Error(
   400→          `WAF 拦截且浏览器策略不可用：${e instanceof Error ? e.message : String(e)}（请安装 Playwright：bun add playwright && bunx playwright install chromium）`
   401→        )
   402→      }
   403→      // 浏览器策略失败后稍候再试一轮 HTTP：此时 jar 里可能已有解题通行 cookie，直连即可通过
   404→      await sleep(1500)
   405→    }
   406→  }
   407→  return { html, status, finalUrl, strategy: 'http', elapsedMs: Date.now() - started, wafChallenged: true }
   408→}

   410→async function fetchWithRetry(
   411→  url: string,
   412→  headers: Record<string, string>,
   413→  timeout: number,
   414→  retries = 2,
   415→  keepUA = false
   416→): Promise<Response> {
   417→  let lastErr: unknown = null
   418→  for (let i = 0; i <= retries; i++) {
   419→    try {
   420→      // 每次重试轮换 UA（模拟多用户）；keepUA=true 时固定 UA（WAF 通行 cookie 与 UA 绑定，换 UA 即失效）
   421→      const res = await fetch(url, {
   422→        headers: i === 0 || keepUA ? headers : { ...headers, 'User-Agent': randomUA() },
   423→        redirect: 'follow',
   424→        signal: AbortSignal.timeout(timeout),
   425→        cache: 'no-store',
   426→      })
   427→      if (!res.ok) {
   428→        // 403（WAF 拒绝）：重试只会加剧封禁，直接抛出
   429→        if (res.status === 403) {
   430→          const body = await res.text().catch(() => '')
   431→          if (isHardDeniedHtml(body)) {
   432→            domainThrottle.reportBlock(url, 180_000)
   433→            throw new Error(`目标站拒绝访问（IP 临时封禁，HTTP 403）：${url}。已自动冷却 3 分钟`)
   434→          }
   435→          lastErr = new Error(`HTTP 403`)
   436→          break
   437→        }
   438→        if (i < retries) {
   439→          // 429/5xx：更长退避（尊重 Retry-After 头）+ 轮换 UA
   440→          lastErr = new Error(`HTTP ${res.status}`)
   441→          let backoff = res.status === 429 ? 1500 + Math.random() * 2000 : 600 + Math.random() * 800
   442→          const ra = Number.parseFloat(res.headers.get('retry-after') ?? '')
   443→          if (Number.isFinite(ra) && ra >= 0) backoff = Math.min(ra * 1000 + 250, 30_000)
   444→          await sleep(backoff)
   445→          continue
   446→        }
   447→        // 最后一次重试仍非 2xx：不再把错误页 body 当解析对象（避免 5xx 错误页文本混入正文）
   448→        throw new Error(`HTTP ${res.status}（${url.slice(0, 120)}）`)
   449→      }
   450→      return res
   451→    } catch (e) {
   452→      if (e instanceof Error && e.message.includes('IP 临时封禁')) throw e
   453→      lastErr = e
   454→      if (i < retries) await sleep(600 + Math.random() * 800)
   455→    }
   456→  }
   457→  throw new Error(`请求失败: ${lastErr instanceof Error ? lastErr.message : String(lastErr)}`)
   458→}

   523→async function fetchWithPlaywright(url: string, cfg: FetchConfig, timeout: number, started: number): Promise<FetchResult> {
   524→  await domainThrottle.wait(url, cfg.throttleGap)
   525→  const browser = await getSharedBrowser()
   526→  if (!browser) {
   527→    throw new Error('Playwright 未安装：请在服务器执行 `bun add playwright && bunx playwright install chromium` 后使用 js 渲染策略')
   528→  }
   529→  const headers = buildHeaders(cfg, url)
   530→  const jarCookie = cookieJar.header(url)
   531→  const context = await browser.newContext({
   532→    userAgent: headers['User-Agent'],
   533→    viewport: { width: 1366, height: 850 },
   534→    locale: 'zh-CN',
   535→    timezoneId: 'Asia/Shanghai',
   536→    extraHTTPHeaders: headers,
   537→  })
   538→  await context.addInitScript(STEALTH_SCRIPT)
   539→
   540→  // 显式 cookies + jar 会话 cookies 一并注入
   541→  const jarCookies: { name: string; value: string; domain?: string }[] = jarCookie
   542→    ? jarCookie.split(';').map((c) => {
   543→        const idx = c.indexOf('=')
   544→        return { name: c.slice(0, idx).trim(), value: c.slice(idx + 1).trim() }
   545→      })
   546→    : []
   547→  const explicitCookies = (cfg.cookies ?? '')
   548→    .split(';')
   549→    .map((c) => c.trim())
   550→    .filter(Boolean)
   551→    .map((c) => {
   552→      const idx = c.indexOf('=')
   553→      if (idx <= 0) return null
   554→      const u = new URL(url)
   555→      return { name: c.slice(0, idx).trim(), value: c.slice(idx + 1).trim(), domain: u.hostname, path: '/' }
   556→    })
   557→    .filter((c): c is { name: string; value: string; domain: string; path: string } => c !== null)
   558→  const allCookies = [...explicitCookies, ...jarCookies].map((c) => ({
   559→    name: c.name,
   560→    value: c.value,
   561→    domain: c.domain ?? new URL(url).hostname,
   562→    path: '/',
   563→  }))
   564→  if (allCookies.length) await context.addCookies(allCookies)
   565→
   566→  try {
   567→    const page = await context.newPage()
   568→    // goto 带一次重试（沙箱/网络偶发 ERR_ADDRESS_UNREACHABLE）
   569→    try {
   570→      await page.goto(url, { waitUntil: 'domcontentloaded', timeout })
   571→    } catch (e) {
   572→      const msg = e instanceof Error ? e.message : String(e)
   573→      if (!/ERR_ADDRESS_UNREACHABLE|ERR_CONNECTION|net::ERR/i.test(msg)) throw e
   574→      await page.waitForTimeout(2500)
   575→      await page.goto(url, { waitUntil: 'domcontentloaded', timeout })
   576→    }
   577→    await page.waitForLoadState('networkidle', { timeout: 6000 }).catch(() => undefined)
   578→    await page.waitForTimeout(800)
   579→
   580→    let html = await page.content()
   581→    let finalUrl = page.url()
   582→
   583→    // WAF 挑战页 → 自动求解验证码
   584→    if (isWafChallengeHtml(html) && !isHardDeniedHtml(html)) {
   585→      const solved = await solveWafChallenge(page)
   586→      if (solved) {
   587→        await page.waitForTimeout(600)
   588→        html = await page.content()
   589→        finalUrl = page.url()
   590→      }
   591→    }
   592→
   593→    // 硬 403（IP 黑名单）：全局冷却后明确抛错，避免静默返回空解析结果
   594→    if (isHardDeniedHtml(html)) {
   595→      domainThrottle.reportBlock(url, 180_000)
   596→      throw new Error(`目标站拒绝浏览器访问（IP 临时封禁，HTTP 403）：${url}。已自动冷却 3 分钟`)
   597→    }
   598→
   599→    // JS 渲染翻页交互：逐项点击分页控件并拼接快照（参数加密/URL 不变站点）
   600→    if (cfg.jsPages?.enabled && cfg.jsPages.itemsSelector && !isWafChallengeHtml(html)) {
   601→      html = await collectJsPages(page, html, cfg.jsPages)
   602→    }
   603→
   604→    // 会话 cookie 回写 jar（WAF 通过后的通行 cookie 全局复用）
   605→    try {
   606→      const cookies = await context.cookies()
   607→      cookieJar.absorbFromBrowser(new URL(url).hostname, cookies)
   608→    } catch {
   609→      /* ignore */
   610→    }
   611→
   612→    return {
   613→      html,
   614→      status: isWafChallengeHtml(html) ? 403 : 200,
   615→      finalUrl,
   616→      strategy: 'playwright',
   617→      elapsedMs: Date.now() - started,
   618→      wafChallenged: isWafChallengeHtml(html),
   619→    }
   620→  } finally {
   621→    // 仅关闭 context（页面级），共享 browser 单例保持存活以复用会话
   622→    await context.close().catch(() => undefined)
   623→  }
   624→}
   625→
   626→/**
   627→ * JS 渲染翻页交互：逐项点击分页控件（下拉/按钮）并拼接每页快照。
   628→ * 适用于分页参数加密、点击后 URL 不变的站点；拼接结果用 <!--JSPAGE--> 分隔，
   629→ * 解析器（css/regex/xpath）对拼接后的 HTML 透明。
   630→ * 点击均用短超时（3s）且单页失败跳过，避免不可见元素 30s 重试导致整体挂起。
   631→ */
   632→async function collectJsPages(
   633→  page: PlaywrightPage,
   634→  firstPageHtml: string,
   635→  js: NonNullable<FetchConfig['jsPages']>
   636→): Promise<string> {
   637→  const snapshots: string[] = [firstPageHtml]
   638→  const CLICK_TIMEOUT = 3000
   639→  try {
   640→    const wait = js.waitAfterClick ?? 1200
   641→    const max = Math.min(js.maxPages ?? 30, 40)
   642→    const startIdx = js.skipFirst ? 1 : 0
   643→    let trigger: PlaywrightElement | null = null
   644→    if (js.triggerSelector) {
   645→      trigger = await page.$(js.triggerSelector)
   646→    }
   647→    for (let i = startIdx; snapshots.length < max; i++) {
   648→      // 每轮重新查询分页项：点击翻页后 DOM 常被重建，跨轮持有的 ElementHandle 会失效（stale element）
   649→      const items = await page.$$(js.itemsSelector)
   650→      if (i >= items.length) break
   651→      try {
   652→        // 下拉类控件需先展开再选目标项
   653→        if (trigger) {
   654→          await trigger.click({ timeout: 1500, force: true }).catch(() => undefined)
   655→          await page.waitForTimeout(200)
   656→        }
   657→        await items[i].click({ timeout: CLICK_TIMEOUT, force: true })
   658→        await page.waitForTimeout(wait)
   659→        await page.waitForLoadState('domcontentloaded', { timeout: 5000 }).catch(() => undefined)
   660→        snapshots.push(await page.content())
   661→      } catch {
   662→        // 单页点击失败跳过
   663→      }
   664→    }
   665→  } catch {
   666→    // 分页控件不存在 → 仅返回首屏
   667→  }
   668→  return snapshots.length > 1 ? snapshots.join('<!--JSPAGE-->') : firstPageHtml
   669→}

   671→/** VLM 识别验证码图片，返回识别出的字符 */
   672→async function recognizeCaptcha(pngBuffer: Buffer): Promise<string> {
   673→  const spec = 'z-ai-web-dev-sdk'
   674→  const mod = (await import(/* webpackIgnore: true */ spec).catch(() => null)) as {
   675→    default: {
   676→      create: () => Promise<{
   677→        chat: {
   678→          completions: {
   679→            createVision: (req: Record<string, unknown>) => Promise<{
   680→              choices?: { message?: { content?: string } }[]
   681→            }>
   682→          }
   683→        }
   684→      }>
   685→    }
   686→  } | null
   687→  if (!mod) throw new Error('z-ai-web-dev-sdk 不可用，无法自动识别验证码')
   688→  const zai = await mod.default.create()
   689→  const base64 = pngBuffer.toString('base64')
   690→  // VLM 调用带 25s 超时保护，避免网络异常时请求挂起
   691→  const response = await Promise.race([
   692→    zai.chat.completions.createVision({
   693→      messages: [
   694→        {
   695→          role: 'user',
   696→          content: [
   697→            {
   698→              type: 'text',
   699→              text: '图片中是一个网站验证码。请准确识别其中的全部字符（可能包含数字与英文字母，注意区分大小写与易混字符如 0/O、1/l/I、6/b、8/B）。只输出识别出的验证码字符本身，不要任何解释、标点或空格。',
   700→            },
   701→            { type: 'image_url', image_url: { url: `data:image/png;base64,${base64}` } },
   702→          ],
   703→        },
   704→      ],
   705→      thinking: { type: 'disabled' },
   706→    }),
   707→    new Promise<never>((_, reject) => setTimeout(() => reject(new Error('VLM 验证码识别超时')), 25000)),
   708→  ])
   709→  const content = response.choices?.[0]?.message?.content ?? ''
   710→  return content.replace(/[^a-zA-Z0-9]/g, '').slice(0, 8)
   711→}

   713→/**
   714→ * GoEdge WAF（及同构验证码页）自动求解：
   715→ * 截图验证码 → VLM 识别 → 填表提交 → 校验是否通过（最多 4 次刷新重试）
   716→ */
   717→async function solveWafChallenge(page: PlaywrightPage): Promise<boolean> {
   718→  for (let attempt = 0; attempt < 4; attempt++) {
   719→    try {
   720→      const img = await page.$('#ui-captcha-image')
   721→      if (!img) return !isWafChallengeHtml(await page.content())
   722→      const shot = await img.screenshot({ encoding: 'base64' })
   723→      const base64 = typeof shot === 'string' ? shot : Buffer.from(shot).toString('base64')
   724→      if (!base64) continue
   725→      const code = await recognizeCaptcha(Buffer.from(base64, 'base64'))
   726→      if (!code || code.length < 3) continue
   727→      const input = await page.$('#GOEDGE_WAF_CAPTCHA_CODE')
   728→      const submit = await page.$('#captcha-form button')
   729→      if (!input || !submit) return !isWafChallengeHtml(await page.content())
   730→      await input.fill(code)
   731→      await submit.click()
   732→      // 等待跳转/刷新（WAF 通过后会 302 回原页面）
   733→      await page.waitForLoadState('domcontentloaded', { timeout: 12000 }).catch(() => undefined)
   734→      await page.waitForTimeout(1500)
   735→      const html = await page.content()
   736→      if (!isWafChallengeHtml(html)) return true
   737→    } catch {
   738→      // 下一轮刷新重试
   739→    }
   740→    // 刷新验证码后重试
   741→    await page.reload({ waitUntil: 'domcontentloaded', timeout: 15000 }).catch(() => undefined)
   742→    await page.waitForTimeout(800)
   743→  }
   744→  return false
   745→}

   824→export async function fetchImage(url: string, referer?: string, timeout = 20000, ua?: string, cookies?: string): Promise<Buffer> {
   825→  await domainThrottle.wait(url)
   826→  const headers: Record<string, string> = {
   827→    'User-Agent': ua ?? randomUA(),
   828→    Referer: referer || new URL(url).origin,
   829→    Accept: 'image/avif,image/webp,image/apng,image/*,*/*;q=0.8',
   830→  }
   831→  const jarCookie = cookies || cookieJar.header(url)
   832→  if (jarCookie) headers.Cookie = jarCookie
   833→  const res = await fetchWithRetry(url, headers, timeout, 1, Boolean(ua))
   834→  const buf = Buffer.from(await res.arrayBuffer())
   835→  if (!res.ok || buf.length === 0) {
   836→    throw new Error(`图片下载失败：HTTP ${res.status}（${url.slice(0, 120)}）`)
   837→  }
   838→  return buf
   839→}
