   114→   114→function presetConfig(type: RuleType): Record<string, unknown> {
   115→   115→  const c = defaultConfig(type) as Record<string, unknown>
   116→   116→  if (type === 'list') {
   117→   117→    const items = c.items as Record<string, FieldSelector>
   118→   118→    items.item = { mode: 'css', expr: '.item, li', attr: 'text' }
   119→   119→    items.title = { mode: 'css', expr: 'h3 a, .title', attr: 'text' }
   120→   120→    items.link = { mode: 'css', expr: 'a', attr: 'href' }
   121→   121→    c.pagination = { enabled: true, mode: 'template', urlTemplate: '', startPage: 1, endPage: 5, maxPages: 20 }
   122→   122→  } else if (type === 'book') {
   123→   123→    const f = c.fields as Record<string, FieldSelector>
   124→   124→    f.title = { mode: 'css', expr: 'h1, .book-title, #title', attr: 'text' }
   125→   125→    f.author = { mode: 'css', expr: '#author, .author', attr: 'text' }
   126→   126→    f.category = { mode: 'css', expr: '.breadcrumb a:nth-child(2), .category', attr: 'text' }
   127→   127→    f.keywords = { mode: 'regex', expr: '<meta\\s+name="keywords"\\s+content="([^"]+)"', group: 1 }
   128→   128→    f.intro = { mode: 'css', expr: '.intro, #intro, .book-desc', attr: 'text' }
   129→   129→    f.cover = { mode: 'css', expr: '.cover img, #cover img', attr: 'src' }
   130→   130→    f.status = { mode: 'css', expr: '.status', attr: 'text' }
   131→   131→    f.latestChapter = { mode: 'css', expr: '.latest a', attr: 'text' }
   132→   132→    f.tocLink = { mode: 'css', expr: 'a:contains("目录"), .btn-toc', attr: 'href' }
   133→   133→  } else if (type === 'toc') {
   134→   134→    const items = c.items as Record<string, FieldSelector>
   135→   135→    items.item = { mode: 'css', expr: '#list dd, .chapter-list li', attr: 'text' }
   136→   136→    items.title = { mode: 'css', expr: 'a', attr: 'text' }
   137→   137→    items.link = { mode: 'css', expr: 'a', attr: 'href' }
   138→   138→    const reorder = c.reorder as { enabled: boolean; numberPattern: string }
   139→   139→    reorder.enabled = true
   140→   140→  } else if (type === 'content') {
   141→   141→    c.content = { mode: 'css', expr: '#content, .read-content, .txt', attr: 'html' }
   142→   142→    c.extraAdPatterns = ['一秒记住.*?', '天才一秒记住.*?']
   143→   143→  }
   144→   144→  return c
   145→   145→}

   343→   343→function TestPanel({ type, config }: { type: RuleType; config: Record<string, unknown> }) {
   344→   344→  const [url, setUrl] = useState('')
   345→   345→  const [running, setRunning] = useState(false)
   346→   346→  const [result, setResult] = useState<TestResult | null>(null)
   347→   347→
   348→   348→  const run = async () => {
   349→   349→    if (!url.trim()) return
   350→   350→    setRunning(true)
   351→   351→    setResult(null)
   352→   352→    try {
   353→   353→      const r = await api<TestResult>('/api/rules/test', {
   354→   354→        method: 'POST',
   355→   355→        body: JSON.stringify({ type, config, url: url.trim() }),
   356→   356→      })
   357→   357→      setResult(r)
   358→   358→    } catch (e) {
   359→   359→      setResult({ ok: false, message: e instanceof Error ? e.message : String(e) })
   360→   360→    } finally {
   361→   361→      setRunning(false)
   362→   362→    }
   363→   363→  }
   364→   364→
   365→   365→  const sample = (result?.data?.sample ?? result?.data?.fields) as unknown
   366→   366→  const preview = String(result?.data?.preview ?? '')
   367→   367→
   368→   368→  return (
   369→   369→    <SubSection title="规则测试（保存前即可验证）">
   370→   370→      <div className="flex gap-2">
   371→   371→        <Input className="h-8 flex-1 font-mono text-xs" placeholder="输入目标页 URL 进行测试"
   372→   372→          value={url} onChange={(e) => setUrl(e.target.value)} />
   373→   373→        <Button size="sm" className="h-8" disabled={running || !url.trim()} onClick={() => void run()}>
   374→   374→          {running ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Play className="h-3.5 w-3.5" />}
   375→   375→          测试
   376→   376→        </Button>
   377→   377→      </div>
   378→   378→      {result && (
   379→   379→        <div className="space-y-2 rounded-md border bg-muted/40 p-2.5">
   380→   380→          <div className="flex items-center gap-2">
   381→   381→            <Badge variant={result.ok ? 'secondary' : 'destructive'}>{result.ok ? '成功' : '失败'}</Badge>
   382→   382→            <span className="text-xs">{result.message}</span>
   383→   383→            {result.elapsedMs !== undefined && <span className="ml-auto text-[10px] text-muted-foreground">{result.elapsedMs}ms</span>}
   384→   384→          </div>
   385→   385→          {Array.isArray(sample) && sample.length > 0 && (
   386→   386→            <ScrollArea className="max-h-44 rounded border bg-background">
   387→   387→              <div className="space-y-1 p-2 font-mono text-[11px] leading-relaxed">

   592→   592→    try {
   593→   593→      const r = await api<{ rules: RuleRow[] }>('/api/rules')

   597→   597→    }
   598→   598→  }, [toast])

   600→   600→  useEffect(() => { void load() }, [load])
   601→   601→
   602→   602→  const rulesOfType = useMemo(() => rules.filter((r) => r.type === tab), [rules, tab])
   603→   603→

   605→   605→    setEditingId(null)
   606→   606→    setName('')
   607→   607→    setConfig(presetConfig(tab))
   608→   608→    setDialogOpen(true)
   609→   609→  }
   610→   610→
   611→   611→  const openEdit = (rule: RuleRow) => {

   615→   615→      setConfig({ ...defaultConfig(rule.type), ...(JSON.parse(rule.config) as Record<string, unknown>) })

   626→   626→    }

   629→   629→      if (editingId) {
   630→   630→        await api(`/api/rules/${editingId}`, { method: 'PUT', body: JSON.stringify({ name, config }) })
   631→   631→      } else {
   632→   632→        await api('/api/rules', { method: 'POST', body: JSON.stringify({ name, type: tab, config }) })
   633→   633→      }
   634→   634→      toast({ title: editingId ? '规则已更新' : '规则已创建' })
