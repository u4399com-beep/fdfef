    47→/** 数字输入防 NaN 注入（输入中间态如 "1e"/"-" 时 Number() 会得到 NaN） */

    49→  const n = Number(raw)

   113→/** 各类型示例规则（一键填充，便于快速理解写法） */
   114→function presetConfig(type: RuleType): Record<string, unknown> {
   115→  const c = defaultConfig(type) as Record<string, unknown>
   116→  if (type === 'list') {
   117→    const items = c.items as Record<string, FieldSelector>
   118→    items.item = { mode: 'css', expr: '.item, li', attr: 'text' }
   119→    items.title = { mode: 'css', expr: 'h3 a, .title', attr: 'text' }
   120→    items.link = { mode: 'css', expr: 'a', attr: 'href' }
   121→    c.pagination = { enabled: true, mode: 'template', urlTemplate: '', startPage: 1, endPage: 5, maxPages: 20 }
   122→  } else if (type === 'book') {
   123→    const f = c.fields as Record<string, FieldSelector>
   124→    f.title = { mode: 'css', expr: 'h1, .book-title, #title', attr: 'text' }
   125→    f.author = { mode: 'css', expr: '#author, .author', attr: 'text' }
   126→    f.category = { mode: 'css', expr: '.breadcrumb a:nth-child(2), .category', attr: 'text' }
   127→    f.keywords = { mode: 'regex', expr: '<meta\\s+name="keywords"\\s+content="([^"]+)"', group: 1 }
   128→    f.intro = { mode: 'css', expr: '.intro, #intro, .book-desc', attr: 'text' }
   129→    f.cover = { mode: 'css', expr: '.cover img, #cover img', attr: 'src' }
   130→    f.status = { mode: 'css', expr: '.status', attr: 'text' }
   131→    f.latestChapter = { mode: 'css', expr: '.latest a', attr: 'text' }
   132→    f.tocLink = { mode: 'css', expr: 'a:contains("目录"), .btn-toc', attr: 'href' }
   133→  } else if (type === 'toc') {
   134→    const items = c.items as Record<string, FieldSelector>
   135→    items.item = { mode: 'css', expr: '#list dd, .chapter-list li', attr: 'text' }
   136→    items.title = { mode: 'css', expr: 'a', attr: 'text' }
   137→    items.link = { mode: 'css', expr: 'a', attr: 'href' }
   138→    const reorder = c.reorder as { enabled: boolean; numberPattern: string }
   139→    reorder.enabled = true
   140→  } else if (type === 'content') {
   141→    c.content = { mode: 'css', expr: '#content, .read-content, .txt', attr: 'html' }
   142→    c.extraAdPatterns = ['一秒记住.*?', '天才一秒记住.*?']
   143→  }
   144→  return c
   145→}

   148→function SelectorRow({
   149→  label, value, onChange, hint,
   150→}: {
   151→  label: string
   152→  value: FieldSelector

   162→      </div>
   163→      <div className="flex flex-wrap items-center gap-1.5">
   164→        <Select value={value.mode} onValueChange={(v) => update({ mode: v as FieldSelector['mode'] })}>
   165→          <SelectTrigger className="h-8 w-[86px] shrink-0 text-xs"><SelectValue /></SelectTrigger>
   166→          <SelectContent>
   167→            <SelectItem value="css">CSS</SelectItem>
   168→            <SelectItem value="regex">正则</SelectItem>
   169→            <SelectItem value="xpath">XPath</SelectItem>
   170→          </SelectContent>
   171→        </Select>

   173→          className="h-8 min-w-[140px] flex-1 font-mono text-xs"

   175→            value.mode === 'css' ? '如 #content 或 .item a' : value.mode === 'regex' ? '如 <div id="c">([\s\S]*?)</div>' : '如 //div[@id="content"]'
   176→          }
   177→          value={value.expr}
   178→          onChange={(e) => update({ expr: e.target.value })}
   179→        />
   180→        {value.mode !== 'regex' && (
   181→          <Select value={value.attr ?? 'text'} onValueChange={(v) => update({ attr: v })}>
   182→            <SelectTrigger className="h-8 w-[92px] shrink-0 text-xs"><SelectValue /></SelectTrigger>

   184→              <SelectItem value="text">text</SelectItem>
   185→              <SelectItem value="html">html</SelectItem>
   186→              <SelectItem value="outerHtml">outerHtml</SelectItem>
   187→              <SelectItem value="href">href</SelectItem>
   188→              <SelectItem value="src">src</SelectItem>
   189→              <SelectItem value="content">content</SelectItem>
   190→            </SelectContent>
   191→          </Select>
   192→        )}
   193→        {value.mode === 'regex' && (
   194→          <Input
   195→            className="h-8 w-16 shrink-0 text-xs"
   196→            type="number" min={0}
   197→            placeholder="组"
   198→            value={value.group ?? 1}

   201→        )}

   203→          <Checkbox checked={!!value.multiple} onCheckedChange={(v) => update({ multiple: v === true })} />
   204→          多值
   205→        </label>
   206→      </div>
   207→    </div>
   208→  )
   209→}

   343→function TestPanel({ type, config }: { type: RuleType; config: Record<string, unknown> }) {
   344→  const [url, setUrl] = useState('')
   345→  const [running, setRunning] = useState(false)
   346→  const [result, setResult] = useState<TestResult | null>(null)
   347→
   348→  const run = async () => {
   349→    if (!url.trim()) return
   350→    setRunning(true)
   351→    setResult(null)
   352→    try {
   353→      const r = await api<TestResult>('/api/rules/test', {
   354→        method: 'POST',
   355→        body: JSON.stringify({ type, config, url: url.trim() }),
   356→      })
   357→      setResult(r)
   358→    } catch (e) {
   359→      setResult({ ok: false, message: e instanceof Error ? e.message : String(e) })
   360→    } finally {
   361→      setRunning(false)
   362→    }
   363→  }
   364→
   365→  const sample = (result?.data?.sample ?? result?.data?.fields) as unknown
   366→  const preview = String(result?.data?.preview ?? '')
   367→
   368→  return (
   369→    <SubSection title="规则测试（保存前即可验证）">
   370→      <div className="flex gap-2">
   371→        <Input className="h-8 flex-1 font-mono text-xs" placeholder="输入目标页 URL 进行测试"
   372→          value={url} onChange={(e) => setUrl(e.target.value)} />
   373→        <Button size="sm" className="h-8" disabled={running || !url.trim()} onClick={() => void run()}>
   374→          {running ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Play className="h-3.5 w-3.5" />}
   375→          测试
   376→        </Button>
   377→      </div>
   378→      {result && (
   379→        <div className="space-y-2 rounded-md border bg-muted/40 p-2.5">
   380→          <div className="flex items-center gap-2">
   381→            <Badge variant={result.ok ? 'secondary' : 'destructive'}>{result.ok ? '成功' : '失败'}</Badge>
   382→            <span className="text-xs">{result.message}</span>
   383→            {result.elapsedMs !== undefined && <span className="ml-auto text-[10px] text-muted-foreground">{result.elapsedMs}ms</span>}
   384→          </div>
   385→          {Array.isArray(sample) && sample.length > 0 && (
   386→            <ScrollArea className="max-h-44 rounded border bg-background">
   387→              <div className="space-y-1 p-2 font-mono text-[11px] leading-relaxed">
   388→                {(sample as unknown[]).slice(0, 30).map((item, i) => (
   389→                  <div key={i} className="truncate border-b border-dashed pb-1 last:border-0">
   390→                    {typeof item === 'object' && item !== null
   391→                      ? Object.entries(item as Record<string, unknown>)
   392→                          .filter(([, v]) => v !== '' && v !== undefined)
   393→                          .map(([k, v]) => `${k}: ${String(v).slice(0, 90)}`)
   394→                          .join(' ｜ ')
   395→                      : String(item).slice(0, 120)}
   396→                  </div>
   397→                ))}
   398→              </div>
   399→            </ScrollArea>
   400→          )}
   401→          {!!sample && !Array.isArray(sample) && (
   402→            <ScrollArea className="max-h-44 rounded border bg-background">
   403→              <div className="space-y-1 p-2 font-mono text-[11px]">
   404→                {Object.entries(sample as Record<string, unknown>).map(([k, v]) => (
   405→                  <div key={k} className="flex gap-2 border-b border-dashed pb-1 last:border-0">
   406→                    <span className="shrink-0 font-medium">{k}:</span>
   407→                    <span className="break-all">{String(v).slice(0, 200)}</span>
   408→                  </div>
   409→                ))}
   410→              </div>
   411→            </ScrollArea>
   412→          )}
   413→          {preview && (
   414→            <pre className="max-h-40 overflow-auto whitespace-pre-wrap rounded border bg-background p-2 text-[11px] leading-relaxed">{preview}</pre>
   415→          )}
   416→          {!!(result.data && (result.data.smartCategory || result.data.smartCompletion)) && (
   417→            <div className="flex flex-wrap gap-2 text-[11px]">
   418→              {result.data.smartCategory !== undefined && (
   419→                <Badge variant="outline">
   420→                  智能分类：{String((result.data.smartCategory as Record<string, unknown>).category)}
   421→                  （置信度 {Number((result.data.smartCategory as Record<string, unknown>).score).toFixed(2)}）
   422→                </Badge>
   423→              )}
   424→              {result.data.smartCompletion !== undefined && (
   425→                <Badge variant="outline">
   426→                  完结判断：{String((result.data.smartCompletion as Record<string, unknown>).status)}
   427→                  （置信度 {Number((result.data.smartCompletion as Record<string, unknown>).confidence).toFixed(2)}）
   428→                </Badge>
   429→              )}
   430→            </div>
   431→          )}
   432→        </div>
   433→      )}
   434→    </SubSection>
   435→  )
   436→}

   438→// ---------------- 规则编辑器 ----------------
   439→function RuleEditor({
   440→  type, ruleId, name, config, onNameChange, onConfigChange,
   441→}: {
   442→  type: RuleType
   443→  ruleId: string | null
   444→  name: string
   445→  config: Record<string, unknown>
   446→  onNameChange: (v: string) => void
   447→  onConfigChange: (c: Record<string, unknown>) => void
   448→}) {
   449→  const patch = (p: Record<string, unknown>) => onConfigChange({ ...config, ...p })
   450→  const setSel = (path: (string | number)[], sel: FieldSelector) => {
   451→    const clone = structuredClone(config) as Record<string, unknown>
   452→    let node: Record<string, unknown> = clone
   453→    for (let i = 0; i < path.length - 1; i++) {
   454→      // 中间节点缺失/非对象时（历史规则数据可能存了 null）自动补建，避免 TypeError
   455→      const next = node[path[i]]
   456→      if (!next || typeof next !== 'object') node[path[i]] = {}
   457→      node = node[path[i]] as Record<string, unknown>
   458→    }
   459→    node[path[path.length - 1]] = sel
   460→    onConfigChange(clone)
   461→  }
   462→
   463→  const items = (config.items ?? {}) as Record<string, FieldSelector>
   464→  const fields = (config.fields ?? {}) as Record<string, FieldSelector>
   465→
   466→  return (

   468→      <div className="space-y-1">
   469→        <Label className="text-xs">规则名称</Label>
   470→        <Input className="h-8" value={name} onChange={(e) => onNameChange(e.target.value)}
   471→          placeholder={`${TYPE_META[type].label}规则`} />
   472→      </div>

   474→      <FetchFields cfg={config} onChange={patch} />
   475→
   476→      {type === 'list' && (

   481→          <SelectorRow label="简介（可选）" value={items.intro ?? emptySelector()} onChange={(s) => setSel(['items', 'intro'], s)} />
   482→          <SelectorRow label="封面（可选）" value={items.cover ?? emptySelector()} onChange={(s) => setSel(['items', 'cover'], s)} />
   483→        </SubSection>
   484→      )}
   485→
   486→      {type === 'book' && (
   487→        <SubSection title="书籍字段解析">
   488→          <SelectorRow label="书名（必填）" value={fields.title ?? emptySelector()} onChange={(s) => setSel(['fields', 'title'], s)} />
   489→          <SelectorRow label="作者" value={fields.author ?? emptySelector()} onChange={(s) => setSel(['fields', 'author'], s)} />
   490→          <SelectorRow label="分类" value={fields.category ?? emptySelector()} onChange={(s) => setSel(['fields', 'category'], s)} hint="留空则智能匹配" />
   491→          <SelectorRow label="关键词" value={fields.keywords ?? emptySelector()} onChange={(s) => setSel(['fields', 'keywords'], s)} />
   492→          <SelectorRow label="简介" value={fields.intro ?? emptySelector()} onChange={(s) => setSel(['fields', 'intro'], s)} />
   493→          <SelectorRow label="封面图" value={fields.cover ?? emptySelector()} onChange={(s) => setSel(['fields', 'cover'], s)} />
   494→          <SelectorRow label="状态（连载/完结）" value={fields.status ?? emptySelector()} onChange={(s) => setSel(['fields', 'status'], s)} />
   495→          <SelectorRow label="最新章节" value={fields.latestChapter ?? emptySelector()} onChange={(s) => setSel(['fields', 'latestChapter'], s)} />
   496→          <SelectorRow label="目录页链接（可选）" value={fields.tocLink ?? emptySelector()} onChange={(s) => setSel(['fields', 'tocLink'], s)} hint="默认用当前页" />
   497→        </SubSection>
   498→      )}
   499→
   500→      {(type === 'list' || type === 'toc') && <PaginationFields cfg={config} onChange={patch} />}
   501→
   502→      {type === 'toc' && (
   503→        <>
   504→          <SubSection title="章节列表解析">
   505→            <SelectorRow label="章节项容器" value={items.item ?? emptySelector()} onChange={(s) => setSel(['items', 'item'], s)} />
   506→            <SelectorRow label="章节标题" value={items.title ?? emptySelector()} onChange={(s) => setSel(['items', 'title'], s)} />
   507→            <SelectorRow label="章节链接" value={items.link ?? emptySelector()} onChange={(s) => setSel(['items', 'link'], s)} />
   508→          </SubSection>
   509→          <SubSection title="乱序重排 / 去重">
   510→            <div className="grid gap-3 md:grid-cols-2">
   511→              <div className="flex items-center justify-between rounded-md border p-2.5">
   512→                <div>
   513→                  <p className="text-sm font-medium">乱序重排</p>
   514→                  <p className="text-[11px] text-muted-foreground">按章节序号重排目录</p>
   515→                </div>
   516→                <Switch
   517→                  checked={!!(config.reorder as Record<string, unknown> | undefined)?.enabled}
   518→                  onCheckedChange={(v) => patch({ reorder: { ...(config.reorder as Record<string, unknown>), enabled: v } })}
   519→                />

   573→      )}
   574→

   581→export function RulesPage() {
   582→  const { toast } = useToast()
   583→  const [rules, setRules] = useState<RuleRow[]>([])
   584→  const [tab, setTab] = useState<RuleType>('list')
   585→  const [dialogOpen, setDialogOpen] = useState(false)
   586→  const [editingId, setEditingId] = useState<string | null>(null)
   587→  const [name, setName] = useState('')
   588→  const [config, setConfig] = useState<Record<string, unknown>>({})
   589→  const [saving, setSaving] = useState(false)
   590→
   591→  const load = useCallback(async () => {
   592→    try {
   593→      const r = await api<{ rules: RuleRow[] }>('/api/rules')
   594→      setRules(r.rules)
   595→    } catch (e) {
   596→      toast({ title: '加载失败', description: e instanceof Error ? e.message : String(e), variant: 'destructive' })
   597→    }
   598→  }, [toast])
   599→
   600→  useEffect(() => { void load() }, [load])
   601→
   602→  const rulesOfType = useMemo(() => rules.filter((r) => r.type === tab), [rules, tab])
   603→
   604→  const openCreate = () => {
   605→    setEditingId(null)
   606→    setName('')
   607→    setConfig(presetConfig(tab))
   608→    setDialogOpen(true)
   609→  }
   610→
   611→  const openEdit = (rule: RuleRow) => {
   612→    setEditingId(rule.id)
   613→    setName(rule.name)
   614→    try {
   615→      setConfig({ ...defaultConfig(rule.type), ...(JSON.parse(rule.config) as Record<string, unknown>) })
   616→    } catch {
   617→      setConfig(defaultConfig(rule.type))
   618→    }
   619→    setDialogOpen(true)
   620→  }
   621→
   622→  const save = async () => {
   623→    if (!name.trim()) {
   624→      toast({ title: '请填写规则名称', variant: 'destructive' })
   625→      return
   626→    }
   627→    setSaving(true)
   628→    try {
   629→      if (editingId) {
   630→        await api(`/api/rules/${editingId}`, { method: 'PUT', body: JSON.stringify({ name, config }) })
   631→      } else {
   632→        await api('/api/rules', { method: 'POST', body: JSON.stringify({ name, type: tab, config }) })
   633→      }
   634→      toast({ title: editingId ? '规则已更新' : '规则已创建' })
   635→      setDialogOpen(false)
   636→      await load()
   637→    } catch (e) {
   638→      toast({ title: '保存失败', description: e instanceof Error ? e.message : String(e), variant: 'destructive' })
   639→    } finally {
   640→      setSaving(false)
   641→    }
   642→  }
   643→
   644→  const remove = async (rule: RuleRow) => {
   645→    if (!confirm(`确定删除规则「${rule.name}」？`)) return
   646→    try {
   647→      await api(`/api/rules/${rule.id}`, { method: 'DELETE' })
   648→      toast({ title: '已删除' })
   649→      await load()
   650→    } catch (e) {
   651→      toast({ title: '删除失败', description: e instanceof Error ? e.message : String(e), variant: 'destructive' })
   652→    }
   653→  }
   654→
   655→  return (
   656→    <div className="space-y-4">
   657→      <div className="flex flex-wrap items-center justify-between gap-3">
   658→        <Tabs value={tab} onValueChange={(v) => setTab(v as RuleType)}>
   659→          <TabsList>
   660→            {(Object.keys(TYPE_META) as RuleType[]).map((t) => {
   661→              const Icon = TYPE_META[t].icon
   662→              return (
   663→                <TabsTrigger key={t} value={t} className="gap-1.5">
   664→                  <Icon className="h-3.5 w-3.5" />
   665→                  {TYPE_META[t].label}
   666→                </TabsTrigger>
   667→              )
   668→            })}
   669→          </TabsList>
   670→        </Tabs>
   671→        <Button onClick={openCreate} size="sm">
   672→          <Plus className="h-4 w-4" /> 新建{TYPE_META[tab].label}规则
   673→        </Button>
   674→      </div>
   675→      <p className="text-xs text-muted-foreground">{TYPE_META[tab].desc} —— 支持 CSS / 正则 / XPath 三种模式混用，每个规则均内置测试。</p>
   676→
   677→      <div className="grid gap-3 md:grid-cols-2 xl:grid-cols-3">
   678→        {rulesOfType.length === 0 && (
   679→          <Card className="md:col-span-2 xl:col-span-3">
   680→            <CardContent className="flex flex-col items-center gap-2 py-10 text-center">
   681→              <Wand2 className="h-8 w-8 text-muted-foreground" />
   682→              <p className="text-sm text-muted-foreground">暂无{TYPE_META[tab].label}规则，点击右上角「新建」（已预填示例结构，可直接改表达式）</p>
   683→            </CardContent>
   684→          </Card>
   685→        )}
   686→        {rulesOfType.map((rule) => (
   687→          <Card key={rule.id}>
   688→            <CardContent className="space-y-2.5 p-4">
   689→              <div className="flex items-start justify-between gap-2">
   690→                <div className="min-w-0">
   691→                  <p className="truncate text-sm font-semibold">{rule.name}</p>
   692→                  <p className="text-[11px] text-muted-foreground">更新于 {new Date(rule.updatedAt).toLocaleString('zh-CN', { hour12: false })}</p>
   693→                </div>
   694→                <Badge variant={rule.enabled ? 'secondary' : 'outline'}>{rule.enabled ? '启用' : '停用'}</Badge>
   695→              </div>
   696→              <div className="flex gap-2">
   697→                <Button size="sm" variant="outline" className="flex-1" onClick={() => openEdit(rule)}>
   698→                  <Pencil className="h-3.5 w-3.5" /> 编辑
   699→                </Button>
   700→                <Button size="sm" variant="outline" className="text-red-600 hover:text-red-700" onClick={() => void remove(rule)} aria-label={`删除规则「${rule.name}」`}>
   701→                  <Trash2 className="h-3.5 w-3.5" />
   702→                </Button>
   703→              </div>
   704→            </CardContent>
   705→          </Card>
   706→        ))}
   707→      </div>
   708→
   709→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   710→        <DialogContent className="max-h-[88vh] overflow-y-auto sm:max-w-2xl">
   711→          <DialogHeader>
   712→            <DialogTitle>{editingId ? '编辑' : '新建'}{TYPE_META[tab].label}规则</DialogTitle>
   713→            <DialogDescription>三种选择器模式可混用；CSS 取属性需选择 attr；正则建议使用捕获组。</DialogDescription>
   714→          </DialogHeader>
   715→          <Separator />
   716→          <RuleEditor
   717→            type={tab}
   718→            ruleId={editingId}
   719→            name={name}
   720→            config={config}
   721→            onNameChange={setName}
   722→            onConfigChange={setConfig}
   723→          />
   724→          <DialogFooter>
   725→            <Button variant="outline" onClick={() => setConfig(presetConfig(tab))}>
   726→              <Beaker className="h-4 w-4" /> 重置为示例
   727→            </Button>
   728→            <Button onClick={() => void save()} disabled={saving}>
   729→              {saving && <Loader2 className="h-4 w-4 animate-spin" />} 保存规则
   730→            </Button>
   731→          </DialogFooter>
   732→        </DialogContent>
   733→      </Dialog>
   734→    </div>
   735→  )
   736→}
