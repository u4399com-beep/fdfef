   446→/* ================= Toc（完整章节目录页） ================= */
   447→
   448→const TOC_PAGE_SIZE = 100
   449→
   450→function InkToc({ site, data, loading, onNavigate }: ThemeProps) {
   451→  const [page, setPage] = useState(1)
   452→  const book = data.book
   453→  const chapters = book?.chapters ?? []
   454→  const totalPages = Math.max(1, Math.ceil(chapters.length / TOC_PAGE_SIZE))
   455→  const safePage = Math.min(page, totalPages)
   456→  const visible = chapters.slice((safePage - 1) * TOC_PAGE_SIZE, safePage * TOC_PAGE_SIZE)
   457→
   458→  return (
   459→    <Shell site={site} poem="千章罗列 · 纲举目张" onNavigate={onNavigate}>
   460→      <div className="mx-auto max-w-4xl">
   461→        <Crumbs
   462→          trail={[
   463→            { label: '首页', onSelect: () => onNavigate({ type: 'home' }) },
   464→            {
   465→              label: book?.title ?? (loading ? '加载中' : '书籍'),
   466→              onSelect: book ? () => onNavigate({ type: 'book', bookId: book.id }) : undefined,
   467→            },
   468→            { label: '章节目录' },
   469→          ]}
   470→        />
   471→
   472→        {loading ? (
   473→          <div className="mt-2">
   474→            <InkBookSkeleton />
   475→          </div>
   476→        ) : !book ? (
   477→          <div className="py-24 text-center">
   478→            <p className="text-stone-500">未寻得此书，或已散佚。</p>
   479→            <button
   480→              type="button"
   481→              onClick={() => onNavigate({ type: 'home' })}
   482→              className="mt-6 min-h-[44px] bg-[#9e2b25] px-6 text-sm tracking-widest text-stone-50 transition-colors hover:bg-[#7f221d]"
   483→            >
   484→              返回首页
   485→            </button>
   486→          </div>
   487→        ) : (
   488→          <>
   489→            <div className="flex flex-wrap items-end justify-between gap-3">
   490→              <h1 className="text-2xl font-bold tracking-widest text-stone-900">
   491→                {book.title}
   492→                <span className="ml-3 text-base font-normal text-stone-400">章节目录</span>
   493→              </h1>
   494→              <p className="text-sm text-stone-500">
   495→                {book.author} · 共 {book.totalChapters} 章
   496→              </p>
   497→            </div>
   498→
   499→            <nav
   500→              aria-label="目录分页"
   501→              className="mt-4 flex flex-wrap items-center gap-3 border-b border-stone-300/70 pb-4"
   502→            >
   503→              <button
   504→                type="button"
   505→                disabled={safePage <= 1}
   506→                onClick={() => setPage((p) => Math.max(1, p - 1))}
   507→                className="inline-flex min-h-[36px] items-center border border-stone-300 bg-white/80 px-4 text-sm tracking-widest text-stone-600 transition-colors hover:border-[#9e2b25] hover:text-[#9e2b25] disabled:cursor-not-allowed disabled:opacity-40"
   508→              >
   509→                上一页
   510→              </button>
   511→              <span className="text-sm text-stone-500">
   512→                第 {safePage} / {totalPages} 页
   513→              </span>
   514→              <button
   515→                type="button"
   516→                disabled={safePage >= totalPages}
   517→                onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   518→                className="inline-flex min-h-[36px] items-center border border-stone-300 bg-white/80 px-4 text-sm tracking-widest text-stone-600 transition-colors hover:border-[#9e2b25] hover:text-[#9e2b25] disabled:cursor-not-allowed disabled:opacity-40"
   519→              >
   520→                下一页
   521→              </button>
   522→            </nav>
   523→
   524→            {visible.length === 0 ? (
   525→              <p className="py-10 text-center text-stone-400">暂无章节。</p>
   526→            ) : (
   527→              <ul className="mb-10 mt-4 grid grid-cols-1 gap-px border border-stone-300/80 bg-stone-200/70 sm:grid-cols-2 lg:grid-cols-3">
   528→                {visible.map((ch) => (
   529→                  <ChapterLine key={ch.id} chapter={ch} onNavigate={onNavigate} />
   530→                ))}
   531→              </ul>
   532→            )}
   533→          </>
   534→        )}
   535→      </div>
   536→    </Shell>
   537→  )
   538→}

   540→/* ================= Chapter ================= */
   541→
   542→function InkChapter({ site, data, loading, onNavigate }: ThemeProps) {
   543→  const chapter = data.chapter
   544→  const paragraphs = chapter
   545→    ? chapter.content
   546→        .split('\n')
   547→        .map((p) => p.trim())
   548→        .filter(Boolean)
   549→    : []
   550→
   551→  return (
   552→    <Shell site={site} poem="笔下起风云 · 书中藏日月" onNavigate={onNavigate}>
   553→      <div className="mx-auto max-w-3xl">
   554→        <Crumbs
   555→          trail={[
   556→            { label: '首页', onSelect: () => onNavigate({ type: 'home' }) },
   557→            chapter
   558→              ? {
   559→                  label: chapter.bookTitle,
   560→                  onSelect: () => onNavigate({ type: 'book', bookId: chapter.bookId }),
   561→                }
   562→              : { label: '书籍' },
   563→            { label: chapter?.title ?? '正文' },
   564→          ]}
   565→        />
   566→
   567→        {loading ? (
   568→          <InkChapterSkeleton />
   569→        ) : !chapter ? (
   570→          <div className="py-24 text-center">
   571→            <p className="text-stone-500">未寻得此章。</p>
   572→            <button
   573→              type="button"
   574→              onClick={() => onNavigate({ type: 'home' })}
   575→              className="mt-6 min-h-[44px] bg-[#9e2b25] px-6 text-sm tracking-widest text-stone-50 transition-colors hover:bg-[#7f221d]"
   576→            >
   577→              返回首页
   578→            </button>
   579→          </div>
   580→        ) : (
   581→          <article className="mt-2 rounded-sm border border-stone-300/80 bg-white/60 px-6 py-10 sm:px-12">
   582→            <h1 className="text-center text-3xl font-bold tracking-widest text-stone-900">
   583→              {chapter.title}
   584→            </h1>
   585→            <p className="mt-3 text-center text-sm tracking-widest text-stone-400">
   586→              《{chapter.bookTitle}》 · 第 {chapter.order} 章
   587→            </p>
   588→            <InkRule className="my-8" />
   589→            <div className="space-y-6 text-[17px] leading-8 text-stone-800">
   590→              {paragraphs.map((p, i) => (
   591→                <p key={i} className="indent-8">
   592→                  {p}
   593→                </p>
   594→              ))}
   595→            </div>
   596→            <InkRule className="my-8" />
   597→            <nav
   598→              aria-label="章节导航"
   599→              className="flex flex-col items-stretch gap-3 sm:flex-row sm:justify-center"
   600→            >
   601→              <button
   602→                type="button"
   603→                disabled={!chapter.prevId}
   604→                onClick={() => {
   605→                  if (chapter.prevId) onNavigate({ type: 'chapter', chapterId: chapter.prevId })
   606→                }}
   607→                className="inline-flex min-h-[44px] items-center justify-center gap-2 border border-stone-300 bg-white/80 px-6 text-sm tracking-widest text-stone-700 transition-colors hover:border-[#9e2b25] hover:text-[#9e2b25] disabled:cursor-not-allowed disabled:opacity-40"
   608→              >
   609→                <ArrowLeft className="h-4 w-4" aria-hidden="true" />
   610→                上一章
   611→              </button>
   612→              <button
   613→                type="button"
   614→                onClick={() => onNavigate({ type: 'toc', bookId: chapter.bookId })}
   615→                className="inline-flex min-h-[44px] items-center justify-center gap-2 bg-[#9e2b25] px-6 text-sm tracking-widest text-stone-50 transition-colors hover:bg-[#7f221d]"
   616→              >
   617→                返回目录
   618→              </button>
   619→              <button
   620→                type="button"
   621→                disabled={!chapter.nextId}
   622→                onClick={() => {
   623→                  if (chapter.nextId) onNavigate({ type: 'chapter', chapterId: chapter.nextId })
   624→                }}
   625→                className="inline-flex min-h-[44px] items-center justify-center gap-2 border border-stone-300 bg-white/80 px-6 text-sm tracking-widest text-stone-700 transition-colors hover:border-[#9e2b25] hover:text-[#9e2b25] disabled:cursor-not-allowed disabled:opacity-40"
   626→              >
   627→                下一章
   628→                <ArrowRight className="h-4 w-4" aria-hidden="true" />
   629→              </button>
   630→            </nav>
   631→          </article>
   632→        )}
   633→      </div>
   634→    </Shell>
   635→  )
   636→}

   638→/* ================= Keyword ================= */
   639→
   640→function InkKeyword({
   641→  site,
   642→  data,
   643→  loading,
   644→  onNavigate,
   645→  keyword,
   646→}: ThemeProps & { keyword: string }) {
   647→  const books = data.keywordBooks ?? []
   648→  const goMain = () =>
   649→    onNavigate(site.mainBookId ? { type: 'book', bookId: site.mainBookId } : { type: 'home' })
   650→  const related = Array.from(
   651→    new Set(
   652→      books
   653→        .flatMap((b) => [...b.keywords, ...b.suggestKeywords])
   654→        .filter((k) => k !== keyword),
   655→    ),
   656→  ).slice(0, 12)
   657→
   658→  return (
   659→    <Shell site={site} poem="千卷藏天地 · 一键入主册" onNavigate={onNavigate}>
   660→      <Crumbs
   661→        trail={[
   662→          { label: '首页', onSelect: () => onNavigate({ type: 'home' }) },
   663→          { label: `关键词：${keyword}` },
   664→        ]}
   665→      />
   666→
   667→      {/* 最醒目的主书籍入口 */}
   668→      <button
   669→        type="button"
   670→        onClick={goMain}
   671→        className="flex min-h-[44px] w-full items-center justify-center gap-3 bg-[#9e2b25] px-6 py-4 text-lg font-bold tracking-[0.3em] text-stone-50 shadow-md transition-colors hover:bg-[#7f221d]"
   672→      >
   673→        <BookOpen className="h-5 w-5" aria-hidden="true" />
   674→        进入主书籍信息页
   675→      </button>
   676→
   677→      <section className="py-10 text-center">
   678→        <p className="text-xs tracking-[0.7em] text-[#9e2b25]">关 键 词</p>
   679→        <h1 className="mt-4 text-4xl font-bold tracking-[0.2em] text-stone-900">{keyword}</h1>
   680→        <p className="mt-4 text-stone-500">与《{keyword}》相关的书籍与内容</p>
   681→      </section>
   682→
   683→      {loading ? (
   684→        <InkSkeletonRows rows={3} />
   685→      ) : books.length === 0 ? (
   686→        <p className="py-16 text-center text-stone-500">未觅得相关书卷。</p>
   687→      ) : (
   688→        <div className="space-y-4 pb-4">
   689→          {books.map((book) => (
   690→            <ScrollCard key={book.id} book={book} onNavigate={onNavigate} />
   691→          ))}
   692→        </div>
   693→      )}
   694→
   695→      {related.length > 0 && (
   696→        <section aria-label="相关关键词" className="mt-8 border-t border-stone-300/70 pb-2 pt-6">
   697→          <h2 className="flex items-center gap-2 text-sm tracking-[0.3em] text-stone-400">
   698→            <Brush className="h-4 w-4 text-[#9e2b25]" aria-hidden="true" />
   699→            相关关键词
   700→          </h2>
   701→          <div className="mt-4 flex flex-wrap gap-2">
   702→            {related.map((k) => (
   703→              <button
   704→                key={k}
   705→                type="button"
   706→                onClick={() => onNavigate({ type: 'keyword', keyword: k })}
   707→                className="min-h-[44px] border border-stone-300 bg-white/80 px-3 tracking-wider text-stone-600 transition-colors hover:border-[#9e2b25] hover:text-[#9e2b25]"
   708→              >
   709→                {k}
   710→              </button>
   711→            ))}
   712→          </div>
   713→        </section>
   714→      )}
   715→    </Shell>
   716→  )
   717→}

   719→/* ================= 主组件 ================= */
   720→
   721→export function ThemeInk(props: ThemeProps): ReactElement {
   722→  const { view } = props
   723→  switch (view.type) {
   724→    case 'home':
   725→      return <InkHome {...props} />
   726→    case 'book':
   727→      return <InkBook {...props} />
   728→    case 'toc':
   729→      return <InkToc {...props} />
   730→    case 'chapter':
   731→      return <InkChapter {...props} />
   732→    case 'keyword':
   733→      return <InkKeyword {...props} keyword={view.keyword} />
   734→    default:
   735→      return <InkHome {...props} />
   736→  }
   737→}
   738→
   739→export default ThemeInk
