    37→    37→export function randomInt(min: number, max: number): number {
    38→    38→  if (!Number.isFinite(min) || !Number.isFinite(max)) return min
    39→    39→  const lo = Math.max(0, Math.ceil(min))
    40→    40→  const hi = Math.max(lo, Math.floor(max))
    41→    41→  if (lo === hi) return lo
    42→    42→  return lo + Math.floor(Math.random() * (hi - lo + 1))
    43→    43→}

    63→    63→interface JarCookie {
    64→    64→  name: string
    65→    65→  value: string
    66→    66→  domain?: string
    67→    67→  path?: string
    68→    68→  expires?: number

    71→    71→class CookieJar {
    72→    72→  private store: Map<string, JarCookie[]>
    73→    73→
    74→    74→  constructor(store?: Map<string, JarCookie[]>) {
    75→    75→    this.store = store ?? new Map()
    76→    76→  }
    77→    77→
    78→    78→  /** 解析 Set-Cookie 响应头并入 jar */
    79→    79→  absorbFromFetch(host: string, headers: Headers): void {
    80→    80→    const raw = typeof headers.getSetCookie === 'function' ? headers.getSetCookie() : []
    81→    81→    if (!raw.length) return
    82→    82→    const list = this.store.get(host) ?? []
    83→    83→    for (const line of raw) {
    84→    84→      const [pair, ...attrs] = line.split(';')
    85→    85→      const idx = pair.indexOf('=')
    86→    86→      if (idx <= 0) continue
    87→    87→      const name = pair.slice(0, idx).trim()
    88→    88→      const value = pair.slice(idx + 1).trim()
    89→    89→      const lower = attrs.map((a) => a.trim().toLowerCase())
    90→    90→      const expired = lower.some((a) => a.startsWith('expires=') && new Date(a.slice(8)).getTime() < Date.now())
    91→    91→      const pos = list.findIndex((c) => c.name === name)
    92→    92→      if (expired) {
    93→    93→        if (pos >= 0) list.splice(pos, 1)
    94→    94→        continue
    95→    95→      }
    96→    96→      if (pos >= 0) list.splice(pos, 1)
    97→    97→      list.push({ name, value })
    98→    98→    }
    99→    99→    this.store.set(host, list)
   100→   100→  }
   101→   101→
   102→   102→  /** Playwright context.cookies() 结果并入 jar */
   103→   103→  absorbFromBrowser(originHost: string, cookies: { name: string; value: string; expires?: number }[]): void {
   104→   104→    const list = this.store.get(originHost) ?? []
   105→   105→    for (const c of cookies) {
   106→   106→      if (!c.name) continue
   107→   107→      const pos = list.findIndex((x) => x.name === c.name)
   108→   108→      if (pos >= 0) list.splice(pos, 1)
   109→   109→      if (c.expires && c.expires > 0 && c.expires * 1000 < Date.now()) continue
   110→   110→      list.push({ name: c.name, value: c.value, expires: c.expires })
   111→   111→    }
   112→   112→    this.store.set(originHost, list)
   113→   113→  }
   114→   114→
   115→   115→  /** 取指定 URL 的 Cookie 请求头 */
   116→   116→  header(url: string): string {
   117→   117→    let host = ''
   118→   118→    try {
   119→   119→      host = new URL(url).hostname
   120→   120→    } catch {
   121→   121→      return ''
   122→   122→    }
   123→   123→    // 允许父子域共享（.kelexs.com 场景）：精确 host + 以点开头的域后缀
   124→   124→    const list: JarCookie[] = []
   125→   125→    for (const [key, cookies] of this.store) {
   126→   126→      if (key === host || host.endsWith(key.replace(/^\./, ''))) list.push(...cookies)
   127→   127→    }
   128→   128→    if (!list.length) return ''
   129→   129→    return list.map((c) => `${c.name}=${c.value}`).join('; ')
   130→   130→  }
   131→   131→
   132→   132→  count(): number {
   133→   133→    let n = 0
   134→   134→    for (const cookies of this.store.values()) n += cookies.length
   135→   135→    return n
   136→   136→  }
   137→   137→}

   156→   156→interface ThrottleState {
   157→   157→  queues: Map<string, Promise<void>>
   158→   158→  blockedUntil: Map<string, number>
   159→   159→}
   160→   160→const THROTTLE_STATE_KEY = '__novelThrottleState'
   161→   161→const globalForThrottle = globalThis as unknown as { [THROTTLE_STATE_KEY]?: ThrottleState }
   162→   162→const throttleState: ThrottleState =
   163→   163→  globalForThrottle[THROTTLE_STATE_KEY] ?? { queues: new Map(), blockedUntil: new Map() }
   164→   164→globalForThrottle[THROTTLE_STATE_KEY] = throttleState

   166→   166→class DomainThrottle {
   167→   167→  private queues = throttleState.queues
   168→   168→  private blockedUntil = throttleState.blockedUntil
   169→   169→
   170→   170→  /**
   171→   171→   * 同域串行化：每个请求至少间隔 minGapMs。
   172→   172→   * 若该域处于封禁冷却期（reportBlock），自动等待冷却结束后再入队。
   173→   173→   */
   174→   174→  async wait(url: string, customGap?: number): Promise<void> {
   175→   175→    let host = ''
   176→   176→    try {
   177→   177→      host = new URL(url).hostname
   178→   178→    } catch {
   179→   179→      return
   180→   180→    }
   181→   181→    // 封禁冷却期：等待解除后继续（避免在封禁期继续请求加剧封禁）
   182→   182→    const blockedUntil = this.blockedUntil.get(host) ?? 0
   183→   183→    const now = Date.now()
   184→   184→    if (now < blockedUntil) {
   185→   185→      await sleep(blockedUntil - now + 200)
   186→   186→      this.blockedUntil.delete(host)
   187→   187→    }
   188→   188→    const gap = Math.max(0, customGap ?? 1200)
   189→   189→    if (gap <= 0) return
   190→   190→    const prev = this.queues.get(host) ?? Promise.resolve()
   191→   191→    const next = prev.then(
   192→   192→      () => new Promise<void>((r) => setTimeout(r, gap)),
   193→   193→      () => new Promise<void>((r) => setTimeout(r, gap))
   194→   194→    )
   195→   195→    this.queues.set(host, next)
   196→   196→    await next
   197→   197→  }
   198→   198→
   199→   199→  /** 目标站封禁（硬 403）：设置同域全局冷却期 */
   200→   200→  reportBlock(url: string, ms = 180_000): void {
   201→   201→    try {
   202→   202→      const host = new URL(url).hostname
   203→   203→      const until = Date.now() + ms + Math.random() * 60_000
   204→   204→      const prev = this.blockedUntil.get(host) ?? 0
   205→   205→      if (until > prev) this.blockedUntil.set(host, until)
   206→   206→    } catch {
   207→   207→      /* ignore */
   208→   208→    }
   209→   209→  }
   210→   210→}

   226→   226→/** 判断 HTML 是否为 WAF/验证码挑战页或硬拒绝页（403 黑名单） */
   227→   227→export function isWafChallengeHtml(html: string): boolean {
   228→   228→  if (isHardDeniedHtml(html)) return true
   229→   229→  if (!html || html.length > 20000) {
   230→   230→    // 验证页很小；超长页面再做签名兜底检查（避免正文误判）
   231→   231→    return WAF_SIGNATURES.slice(0, 2).some((re) => re.test(html))
   232→   232→  }
   233→   233→  return WAF_SIGNATURES.some((re) => re.test(html))
   234→   234→}

   236→   236→/** 硬 403/封禁页（IP 黑名单，需向调用方明确报告而非静默解析为空） */
   237→   237→export function isHardDeniedHtml(html: string): boolean {
   238→   238→  if (!html || html.length > 4000) return false
   239→   239→  return /<title>\s*403\s*Forbidden\s*<\/title>/i.test(html) || /\b403 Forbidden\b/.test(html)
   240→   240→}

   246→   246→function detectCharset(buffer: Buffer, contentType: string, fallback: string): string {
   247→   247→  const explicit = fallback !== 'auto' ? fallback : ''
   248→   248→  if (explicit) return explicit
   249→   249→  const ctMatch = /charset=["']?([\w-]+)/i.exec(contentType)
   250→   250→  if (ctMatch) return ctMatch[1].toLowerCase()
   251→   251→  const head = buffer.subarray(0, 4096).toString('latin1')
   252→   252→  const meta = /<meta[^>]+charset=["']?([\w-]+)/i.exec(head) ?? /charset=["']([\w-]+)/i.exec(head)
   253→   253→  if (meta) return meta[1].toLowerCase()
   254→   254→  return 'utf-8'
   255→   255→}
   256→   256→
   257→   257→function decodeBuffer(buffer: Buffer, charset: string): string {
   258→   258→  const cs = charset.toLowerCase()
   259→   259→  if (cs === 'utf-8' || cs === 'utf8' || cs === 'ascii') {
   260→   260→    return new TextDecoder('utf-8').decode(buffer)
   261→   261→  }
   262→   262→  // 动态加载 iconv-lite 以处理 gbk/gb2312/big5
   263→   263→  try {
   264→   264→    // eslint-disable-next-line @typescript-eslint/no-require-imports
   265→   265→    const iconv = require('iconv-lite') as {
   266→   266→      decode: (b: Buffer, enc: string) => string
   267→   267→      encodingExists: (enc: string) => boolean
   268→   268→    }
   269→   269→    if (iconv.encodingExists(cs === 'gb2312' ? 'gbk' : cs)) {
   270→   270→      return iconv.decode(buffer, cs === 'gb2312' ? 'gbk' : cs)
   271→   271→    }
   272→   272→  } catch {
   273→   273→    /* fallthrough */
   274→   274→  }
   275→   275→  return new TextDecoder('utf-8').decode(buffer)
   276→   276→}

   282→   282→function buildHeaders(cfg: FetchConfig, url: string, uaOverride?: string): Record<string, string> {
   283→   283→  const ua = uaOverride ?? (cfg.rotateUA === false ? UA_LIST[0] : randomUA())
   284→   284→  const headers: Record<string, string> = {
   285→   285→    'User-Agent': ua,
   286→   286→    Accept: 'text/html,application/xhtml+xml,application/xml;q=0.9,image/avif,image/webp,*/*;q=0.8',
   287→   287→    'Accept-Language': 'zh-CN,zh;q=0.9,en;q=0.6',
   288→   288→    'Cache-Control': 'no-cache',
   289→   289→    Pragma: 'no-cache',
   290→   290→    'Upgrade-Insecure-Requests': '1',
   291→   291→  }
   292→   292→  // Chrome 系 UA 注入配套 Client Hints（真实浏览器指纹）
   293→   293→  const chromeVer = chromeMajorVersion(ua)
   294→   294→  if (chromeVer) {
   295→   295→    headers['sec-ch-ua'] = `"Chromium";v="${chromeVer}", "Google Chrome";v="${chromeVer}", "Not-A.Brand";v="99"`
   296→   296→    headers['sec-ch-ua-mobile'] = /Mobile|Android|iPhone/.test(ua) ? '?1' : '?0'
   297→   297→    headers['sec-ch-ua-platform'] = /Windows/i.test(ua)
   298→   298→      ? '"Windows"'
   299→   299→      : /Macintosh/i.test(ua)
   300→   300→        ? '"macOS"'
   301→   301→        : /Android/i.test(ua)
   302→   302→          ? '"Android"'
   303→   303→          : /iPhone|iPad/i.test(ua)
   304→   304→            ? '"iOS"'
   305→   305→            : '"Linux"'
   306→   306→  }
   307→   307→  if (cfg.referer) headers.Referer = cfg.referer
   308→   308→  if (cfg.cookies) headers.Cookie = cfg.cookies
   309→   309→  // 会话 cookie 自动附带（用户显式 cookies 优先）
   310→   310→  if (!cfg.cookies) {
   311→   311→    const jarCookie = cookieJar.header(url)
   312→   312→    if (jarCookie) headers.Cookie = jarCookie
   313→   313→  }
   314→   314→  if (cfg.headers) Object.assign(headers, cfg.headers)
   315→   315→  return headers
   316→   316→}

   318→   318→/** 主入口：按策略抓取页面 */
   319→   319→export async function fetchPage(url: string, cfg: FetchConfig = {}): Promise<FetchResult> {
   320→   320→  if (!/^https?:\/\//i.test(url)) throw new Error(`非法 URL：${url}`)
   321→   321→  const strategy = cfg.strategy ?? 'http'
   322→   322→  const timeout = cfg.timeout ?? 20000
   323→   323→  const started = Date.now()
   324→   324→
   325→   325→  if (strategy === 'playwright') return fetchWithPlaywright(url, cfg, timeout, started)
   326→   326→  if (strategy === 'hyperbrowser') return fetchWithHyperbrowser(url, cfg, timeout, started)
   327→   327→
   328→   328→  // HTTP 直连策略（遇 WAF 自动升级浏览器渲染；同域全局节流）
   329→   329→  await domainThrottle.wait(url, cfg.throttleGap)
   330→   330→  let html = ''
   331→   331→  let status = 0
   332→   332→  let finalUrl = url
   333→   333→  for (let attempt = 0; attempt < 2; attempt++) {
   334→   334→    const res = await fetchWithRetry(url, buildHeaders(cfg, url), timeout)
   335→   335→    const buffer = Buffer.from(await res.arrayBuffer())
   336→   336→    cookieJar.absorbFromFetch(new URL(res.url || url).hostname, res.headers)
   337→   337→    const charset = detectCharset(buffer, res.headers.get('content-type') ?? '', cfg.encoding ?? 'auto')
   338→   338→    html = decodeBuffer(buffer, charset)
   339→   339→    status = res.status
   340→   340→    finalUrl = res.url || url
   341→   341→    if (isHardDeniedHtml(html)) {
   342→   342→      // IP 已被目标站拉黑（硬 403）：全局冷却后向调用方明确报告
   343→   343→      domainThrottle.reportBlock(finalUrl, 180_000)
   344→   344→      throw new Error(`目标站拒绝访问（IP 临时封禁，HTTP 403）：${finalUrl}。已自动冷却 3 分钟`)
   345→   345→    }
   346→   346→    if (!isWafChallengeHtml(html)) {
   347→   347→      return { html, status, finalUrl, strategy: 'http', elapsedMs: Date.now() - started }
   348→   348→    }
   349→   349→    // WAF 拦截：升级为 Playwright（带验证码求解），仅尝试一次
   350→   350→    try {
   351→   351→      const pwResult = await fetchWithPlaywright(url, cfg, timeout, started)
   352→   352→      return { ...pwResult, elapsedMs: Date.now() - started }
   353→   353→    } catch (e) {
   354→   354→      if (attempt === 1) {
   355→   355→        throw new Error(
   356→   356→          `WAF 拦截且浏览器策略不可用：${e instanceof Error ? e.message : String(e)}（请安装 Playwright：bun add playwright && bunx playwright install chromium）`
   357→   357→        )
   358→   358→      }
   359→   359→    }
   360→   360→  }
   361→   361→  return { html, status, finalUrl, strategy: 'http', elapsedMs: Date.now() - started, wafChallenged: true }
   362→   362→}

   364→   364→async function fetchWithRetry(url: string, headers: Record<string, string>, timeout: number, retries = 2): Promise<Response> {
   365→   365→  let lastErr: unknown = null
   366→   366→  for (let i = 0; i <= retries; i++) {
   367→   367→    try {
   368→   368→      // 每次重试轮换 UA（模拟多用户）
   369→   369→      const res = await fetch(url, {
   370→   370→        headers: i === 0 ? headers : { ...headers, 'User-Agent': randomUA() },
   371→   371→        redirect: 'follow',
   372→   372→        signal: AbortSignal.timeout(timeout),
   373→   373→        cache: 'no-store',
   374→   374→      })
   375→   375→      if (!res.ok) {
   376→   376→        // 403（WAF 拒绝）：重试只会加剧封禁，直接抛出
   377→   377→        if (res.status === 403) {
   378→   378→          const body = await res.text().catch(() => '')
   379→   379→          if (isHardDeniedHtml(body)) {
   380→   380→            domainThrottle.reportBlock(url, 180_000)
   381→   381→            throw new Error(`目标站拒绝访问（IP 临时封禁，HTTP 403）：${url}。已自动冷却 3 分钟`)
   382→   382→          }
   383→   383→          lastErr = new Error(`HTTP 403`)
   384→   384→          break
   385→   385→        }
   386→   386→        if (i < retries) {
   387→   387→          // 429/5xx：更长退避 + 轮换 UA
   388→   388→          lastErr = new Error(`HTTP ${res.status}`)
   389→   389→          await sleep(res.status === 429 ? 1500 + Math.random() * 2000 : 600 + Math.random() * 800)
   390→   390→          continue
   391→   391→        }
   392→   392→      }
   393→   393→      return res
   394→   394→    } catch (e) {
   395→   395→      if (e instanceof Error && e.message.includes('IP 临时封禁')) throw e
   396→   396→      lastErr = e
   397→   397→      if (i < retries) await sleep(600 + Math.random() * 800)
   398→   398→    }
   399→   399→  }
   400→   400→  throw new Error(`请求失败: ${lastErr instanceof Error ? lastErr.message : String(lastErr)}`)
   401→   401→}

   466→   466→async function fetchWithPlaywright(url: string, cfg: FetchConfig, timeout: number, started: number): Promise<FetchResult> {
   467→   467→  await domainThrottle.wait(url, cfg.throttleGap)
   468→   468→  const browser = await getSharedBrowser()
   469→   469→  if (!browser) {
   470→   470→    throw new Error('Playwright 未安装：请在服务器执行 `bun add playwright && bunx playwright install chromium` 后使用 js 渲染策略')
   471→   471→  }
   472→   472→  const headers = buildHeaders(cfg, url)
   473→   473→  const jarCookie = cookieJar.header(url)
   474→   474→  const context = await browser.newContext({
   475→   475→    userAgent: headers['User-Agent'],
   476→   476→    viewport: { width: 1366, height: 850 },
   477→   477→    locale: 'zh-CN',
   478→   478→    timezoneId: 'Asia/Shanghai',
   479→   479→    extraHTTPHeaders: headers,
   480→   480→  })
   481→   481→  await context.addInitScript(STEALTH_SCRIPT)
   482→   482→
   483→   483→  // 显式 cookies + jar 会话 cookies 一并注入
   484→   484→  const jarCookies: { name: string; value: string; domain?: string }[] = jarCookie
   485→   485→    ? jarCookie.split(';').map((c) => {
   486→   486→        const idx = c.indexOf('=')
   487→   487→        return { name: c.slice(0, idx).trim(), value: c.slice(idx + 1).trim() }
   488→   488→      })
   489→   489→    : []
   490→   490→  const explicitCookies = (cfg.cookies ?? '')
   491→   491→    .split(';')
   492→   492→    .map((c) => c.trim())
   493→   493→    .filter(Boolean)
   494→   494→    .map((c) => {
   495→   495→      const idx = c.indexOf('=')
   496→   496→      if (idx <= 0) return null
   497→   497→      const u = new URL(url)
   498→   498→      return { name: c.slice(0, idx).trim(), value: c.slice(idx + 1).trim(), domain: u.hostname, path: '/' }
   499→   499→    })
   500→   500→    .filter((c): c is { name: string; value: string; domain: string; path: string } => c !== null)
   501→   501→  const allCookies = [...explicitCookies, ...jarCookies].map((c) => ({
   502→   502→    name: c.name,
   503→   503→    value: c.value,
   504→   504→    domain: c.domain ?? new URL(url).hostname,
   505→   505→    path: '/',
   506→   506→  }))
   507→   507→  if (allCookies.length) await context.addCookies(allCookies)
   508→   508→
   509→   509→  try {
   510→   510→    const page = await context.newPage()
   511→   511→    // goto 带一次重试（沙箱/网络偶发 ERR_ADDRESS_UNREACHABLE）
   512→   512→    try {
   513→   513→      await page.goto(url, { waitUntil: 'domcontentloaded', timeout })
   514→   514→    } catch (e) {
   515→   515→      const msg = e instanceof Error ? e.message : String(e)
   516→   516→      if (!/ERR_ADDRESS_UNREACHABLE|ERR_CONNECTION|net::ERR/i.test(msg)) throw e
   517→   517→      await page.waitForTimeout(2500)
   518→   518→      await page.goto(url, { waitUntil: 'domcontentloaded', timeout })
   519→   519→    }
   520→   520→    await page.waitForLoadState('networkidle', { timeout: 6000 }).catch(() => undefined)
   521→   521→    await page.waitForTimeout(800)
   522→   522→
   523→   523→    let html = await page.content()
   524→   524→    let finalUrl = page.url()
   525→   525→
   526→   526→    // WAF 挑战页 → 自动求解验证码
   527→   527→    if (isWafChallengeHtml(html) && !isHardDeniedHtml(html)) {
   528→   528→      const solved = await solveWafChallenge(page)
   529→   529→      if (solved) {
   530→   530→        await page.waitForTimeout(600)
   531→   531→        html = await page.content()
   532→   532→        finalUrl = page.url()
   533→   533→      }
   534→   534→    }
   535→   535→
   536→   536→    // 硬 403（IP 黑名单）：全局冷却后明确抛错，避免静默返回空解析结果
   537→   537→    if (isHardDeniedHtml(html)) {
   538→   538→      domainThrottle.reportBlock(url, 180_000)
   539→   539→      throw new Error(`目标站拒绝浏览器访问（IP 临时封禁，HTTP 403）：${url}。已自动冷却 3 分钟`)
   540→   540→    }
   541→   541→
   542→   542→    // JS 渲染翻页交互：逐项点击分页控件并拼接快照（参数加密/URL 不变站点）
   543→   543→    if (cfg.jsPages?.enabled && cfg.jsPages.itemsSelector && !isWafChallengeHtml(html)) {
   544→   544→      html = await collectJsPages(page, html, cfg.jsPages)
   545→   545→    }
   546→   546→
   547→   547→    // 会话 cookie 回写 jar（WAF 通过后的通行 cookie 全局复用）
   548→   548→    try {
   549→   549→      const cookies = await context.cookies()
   550→   550→      cookieJar.absorbFromBrowser(new URL(url).hostname, cookies)
   551→   551→    } catch {
   552→   552→      /* ignore */
   553→   553→    }
   554→   554→
   555→   555→    return {
   556→   556→      html,
   557→   557→      status: isWafChallengeHtml(html) ? 403 : 200,
   558→   558→      finalUrl,
   559→   559→      strategy: 'playwright',
   560→   560→      elapsedMs: Date.now() - started,
   561→   561→      wafChallenged: isWafChallengeHtml(html),
   562→   562→    }
   563→   563→  } finally {
   564→   564→    // 仅关闭 context（页面级），共享 browser 单例保持存活以复用会话
   565→   565→    await context.close().catch(() => undefined)
   566→   566→  }
   567→   567→}
   568→   568→
   569→   569→/**
   570→   570→ * JS 渲染翻页交互：逐项点击分页控件（下拉/按钮）并拼接每页快照。
   571→   571→ * 适用于分页参数加密、点击后 URL 不变的站点；拼接结果用 <!--JSPAGE--> 分隔，
   572→   572→ * 解析器（css/regex/xpath）对拼接后的 HTML 透明。
   573→   573→ * 点击均用短超时（3s）且单页失败跳过，避免不可见元素 30s 重试导致整体挂起。
   574→   574→ */
   575→   575→async function collectJsPages(
   576→   576→  page: PlaywrightPage,
   577→   577→  firstPageHtml: string,
   578→   578→  js: NonNullable<FetchConfig['jsPages']>
   579→   579→): Promise<string> {
   580→   580→  const snapshots: string[] = [firstPageHtml]
   581→   581→  const CLICK_TIMEOUT = 3000
   582→   582→  try {
   583→   583→    const items = await page.$$(js.itemsSelector)
   584→   584→    const wait = js.waitAfterClick ?? 1200
   585→   585→    const max = Math.min(js.maxPages ?? 30, 40)
   586→   586→    const startIdx = js.skipFirst ? 1 : 0
   587→   587→    let trigger: PlaywrightElement | null = null
   588→   588→    if (js.triggerSelector) {
   589→   589→      trigger = await page.$(js.triggerSelector)
   590→   590→    }
   591→   591→    for (let i = startIdx; i < items.length && snapshots.length < max; i++) {
   592→   592→      try {
   593→   593→        // 下拉类控件需先展开再选目标项
   594→   594→        if (trigger) {
   595→   595→          await trigger.click({ timeout: 1500, force: true }).catch(() => undefined)
   596→   596→          await page.waitForTimeout(200)
   597→   597→        }
   598→   598→        await items[i].click({ timeout: CLICK_TIMEOUT, force: true })
   599→   599→        await page.waitForTimeout(wait)
   600→   600→        await page.waitForLoadState('domcontentloaded', { timeout: 5000 }).catch(() => undefined)
   601→   601→        snapshots.push(await page.content())
   602→   602→      } catch {
   603→   603→        // 单页点击失败跳过
   604→   604→      }
   605→   605→    }
   606→   606→  } catch {
   607→   607→    // 分页控件不存在 → 仅返回首屏
   608→   608→  }
   609→   609→  return snapshots.length > 1 ? snapshots.join('<!--JSPAGE-->') : firstPageHtml
   610→   610→}

   613→   613→async function recognizeCaptcha(pngBuffer: Buffer): Promise<string> {
   614→   614→  const spec = 'z-ai-web-dev-sdk'
   615→   615→  const mod = (await import(/* webpackIgnore: true */ spec).catch(() => null)) as {
   616→   616→    default: {
   617→   617→      create: () => Promise<{
   618→   618→        chat: {
   619→   619→          completions: {
   620→   620→            createVision: (req: Record<string, unknown>) => Promise<{
   621→   621→              choices?: { message?: { content?: string } }[]
   622→   622→            }>
   623→   623→          }
   624→   624→        }
   625→   625→      }>
   626→   626→    }
   627→   627→  } | null
   628→   628→  if (!mod) throw new Error('z-ai-web-dev-sdk 不可用，无法自动识别验证码')
   629→   629→  const zai = await mod.default.create()
   630→   630→  const base64 = pngBuffer.toString('base64')
   631→   631→  // VLM 调用带 25s 超时保护，避免网络异常时请求挂起
   632→   632→  const response = await Promise.race([
   633→   633→    zai.chat.completions.createVision({
   634→   634→      messages: [
   635→   635→        {
   636→   636→          role: 'user',
   637→   637→          content: [
   638→   638→            {
   639→   639→              type: 'text',
   640→   640→              text: '图片中是一个网站验证码。请准确识别其中的全部字符（可能包含数字与英文字母，注意区分大小写与易混字符如 0/O、1/l/I、6/b、8/B）。只输出识别出的验证码字符本身，不要任何解释、标点或空格。',
   641→   641→            },
   642→   642→            { type: 'image_url', image_url: { url: `data:image/png;base64,${base64}` } },
   643→   643→          ],
   644→   644→        },
   645→   645→      ],
   646→   646→      thinking: { type: 'disabled' },
   647→   647→    }),
   648→   648→    new Promise<never>((_, reject) => setTimeout(() => reject(new Error('VLM 验证码识别超时')), 25000)),
   649→   649→  ])
   650→   650→  const content = response.choices?.[0]?.message?.content ?? ''
   651→   651→  return content.replace(/[^a-zA-Z0-9]/g, '').slice(0, 8)
   652→   652→}

   654→   654→/**
   655→   655→ * GoEdge WAF（及同构验证码页）自动求解：
   656→   656→ * 截图验证码 → VLM 识别 → 填表提交 → 校验是否通过（最多 4 次刷新重试）
   657→   657→ */
   658→   658→async function solveWafChallenge(page: PlaywrightPage): Promise<boolean> {
   659→   659→  for (let attempt = 0; attempt < 4; attempt++) {
   660→   660→    try {
   661→   661→      const img = await page.$('#ui-captcha-image')
   662→   662→      if (!img) return !isWafChallengeHtml(await page.content())
   663→   663→      const shot = await img.screenshot({ encoding: 'base64' })
   664→   664→      const base64 = typeof shot === 'string' ? shot : Buffer.from(shot).toString('base64')
   665→   665→      if (!base64) continue
   666→   666→      const code = await recognizeCaptcha(Buffer.from(base64, 'base64'))
   667→   667→      if (!code || code.length < 3) continue
   668→   668→      const input = await page.$('#GOEDGE_WAF_CAPTCHA_CODE')
   669→   669→      const submit = await page.$('#captcha-form button')
   670→   670→      if (!input || !submit) return !isWafChallengeHtml(await page.content())
   671→   671→      await input.fill(code)
   672→   672→      await submit.click()
   673→   673→      // 等待跳转/刷新（WAF 通过后会 302 回原页面）
   674→   674→      await page.waitForLoadState('domcontentloaded', { timeout: 12000 }).catch(() => undefined)
   675→   675→      await page.waitForTimeout(1500)
   676→   676→      const html = await page.content()
   677→   677→      if (!isWafChallengeHtml(html)) return true
   678→   678→    } catch {
   679→   679→      // 下一轮刷新重试
   680→   680→    }
   681→   681→    // 刷新验证码后重试
   682→   682→    await page.reload({ waitUntil: 'domcontentloaded', timeout: 15000 }).catch(() => undefined)
   683→   683→    await page.waitForTimeout(800)
   684→   684→  }
   685→   685→  return false
   686→   686→}
