   461→/* ================= Chapter ================= */
   462→
   463→function NeonChapter({ site, data, loading, onNavigate }: ThemeProps) {
   464→  const chapter = data.chapter
   465→  const paragraphs = chapter
   466→    ? chapter.content
   467→        .split('\n')
   468→        .map((p) => p.trim())
   469→        .filter(Boolean)
   470→    : []
   471→
   472→  return (
   473→    <Shell site={site} onNavigate={onNavigate}>
   474→      <Crumbs
   475→        trail={[
   476→          { label: '首页', onSelect: () => onNavigate({ type: 'home' }) },
   477→          chapter
   478→            ? {
   479→                label: chapter.bookTitle,
   480→                onSelect: () => onNavigate({ type: 'book', bookId: chapter.bookId }),
   481→              }
   482→            : { label: '书籍' },
   483→          { label: chapter?.title ?? '正文' },
   484→        ]}
   485→      />
   486→
   487→      {loading ? (
   488→        <ChapterSkeleton />
   489→      ) : !chapter ? (
   490→        <div className="py-24 text-center">
   491→          <p className="text-sm text-zinc-500">未找到该章节内容。</p>
   492→          <button
   493→            type="button"
   494→            onClick={() => onNavigate({ type: 'home' })}
   495→            className="mt-6 inline-flex min-h-[44px] items-center rounded-full bg-white/10 px-6 text-sm font-bold text-zinc-100 transition-colors hover:bg-white/20"
   496→          >
   497→            返回首页
   498→          </button>
   499→        </div>
   500→      ) : (
   501→        <>
   502→          <article className="py-4">
   503→            <p className="font-mono text-xs font-bold uppercase tracking-[0.4em] text-fuchsia-400">
   504→              Chapter {String(chapter.order).padStart(3, '0')}
   505→            </p>
   506→            <h1 className="mt-3 text-3xl font-black tracking-tight text-white sm:text-4xl">
   507→              {chapter.title}
   508→            </h1>
   509→            <p className="mt-3 text-sm text-zinc-500">《{chapter.bookTitle}》</p>
   510→            <div className={`mt-6 h-1 w-24 rounded-full ${GRAD_R}`} aria-hidden="true" />
   511→            <div className="mt-8 max-w-3xl space-y-5">
   512→              {paragraphs.map((p, i) => (
   513→                <p key={i} className="text-[17px] leading-8 text-zinc-200">

   515→                </p>
   516→              ))}
   517→            </div>
   518→          </article>
   519→          <nav
   520→            aria-label="章节导航"
   521→            className="mt-8 flex flex-col gap-3 pb-4 sm:flex-row sm:justify-between"

   529→              className="inline-flex min-h-[44px] items-center justify-center gap-2 rounded-full border border-white/15 px-6 text-sm font-semibold text-zinc-300 transition-colors hover:border-fuchsia-400/60 hover:text-white disabled:cursor-not-allowed disabled:opacity-40"

   535→              type="button"

   540→            </button>

   542→              type="button"

   545→                if (chapter.nextId) onNavigate({ type: 'chapter', chapterId: chapter.nextId })

   549→              下一章
   550→              <ArrowRight className="h-4 w-4" aria-hidden="true" />
   551→            </button>

   559→/* ================= Toc（完整章节目录页） ================= */
   560→
   561→const TOC_PAGE_SIZE = 100
   562→
   563→function NeonToc({ site, data, loading, onNavigate }: ThemeProps) {
   564→  const [page, setPage] = useState(1)
   565→  const book = data.book
   566→  const chapters = book?.chapters ?? []
   567→  const totalPages = Math.max(1, Math.ceil(chapters.length / TOC_PAGE_SIZE))
   568→  const safePage = Math.min(page, totalPages)
   569→  const visible = chapters.slice((safePage - 1) * TOC_PAGE_SIZE, safePage * TOC_PAGE_SIZE)
   570→
   571→  return (
   572→    <Shell site={site} onNavigate={onNavigate}>
   573→      <Crumbs
   574→        trail={[
   575→          { label: '首页', onSelect: () => onNavigate({ type: 'home' }) },
   576→          {
   577→            label: book?.title ?? '书籍',
   578→            onSelect: book ? () => onNavigate({ type: 'book', bookId: book.id }) : undefined,
   579→          },
   580→          { label: '章节目录' },
   581→        ]}
   582→      />
   583→
   584→      {loading ? (
   585→        <BookSkeleton />
   586→      ) : !book ? (
   587→        <div className="py-24 text-center">
   588→          <p className="text-sm text-zinc-500">未找到该书籍。</p>
   589→          <button
   590→            type="button"
   591→            onClick={() => onNavigate({ type: 'home' })}
   592→            className="mt-6 inline-flex min-h-[44px] items-center rounded-full bg-white/10 px-6 text-sm font-bold text-zinc-100 transition-colors hover:bg-white/20"
   593→          >
   594→            返回首页
   595→          </button>
   596→        </div>
   597→      ) : (
   598→        <>
   599→          <p className="font-mono text-xs font-bold uppercase tracking-[0.4em] text-zinc-500">
   600→            Chapters
   601→          </p>
   602→          <div className="mt-3 flex flex-wrap items-end justify-between gap-3">
   603→            <h1 className="text-2xl font-black tracking-tight text-white sm:text-3xl">
   604→              {book.title}
   605→              <span className="ml-3 text-base font-medium text-zinc-500">章节目录</span>
   606→            </h1>
   607→            <p className="text-sm text-zinc-500">
   608→              {book.author} · 共 {book.totalChapters} 章
   609→            </p>
   610→          </div>
   611→
   612→          <nav
   613→            aria-label="目录分页"
   614→            className="mt-5 flex flex-wrap items-center gap-2 rounded-2xl bg-white/5 px-4 py-3 ring-1 ring-white/10"
   615→          >
   616→            <button
   617→              type="button"
   618→              disabled={safePage <= 1}
   619→              onClick={() => setPage((p) => Math.max(1, p - 1))}
   620→              className="min-h-[36px] rounded-full border border-white/15 px-4 text-sm font-semibold text-zinc-300 transition-colors hover:border-fuchsia-400/60 hover:text-white disabled:cursor-not-allowed disabled:opacity-40"
   621→            >
   622→              上一页
   623→            </button>
   624→            <span className="text-sm text-zinc-500">
   625→              第 {safePage} / {totalPages} 页
   626→            </span>
   627→            <button
   628→              type="button"
   629→              disabled={safePage >= totalPages}
   630→              onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   631→              className="min-h-[36px] rounded-full border border-white/15 px-4 text-sm font-semibold text-zinc-300 transition-colors hover:border-fuchsia-400/60 hover:text-white disabled:cursor-not-allowed disabled:opacity-40"
   632→            >
   633→              下一页
   634→            </button>
   635→          </nav>
   636→
   637→          {visible.length === 0 ? (
   638→            <p className="py-10 text-center text-sm text-zinc-500">暂无章节。</p>
   639→          ) : (
   640→            <ul className="mb-2 mt-4 grid grid-cols-1 gap-1 rounded-2xl bg-black/30 p-2 ring-1 ring-white/10 sm:grid-cols-2 lg:grid-cols-3">
   641→              {visible.map((ch) => (
   642→                <ChapterRow key={ch.id} chapter={ch} onNavigate={onNavigate} />
   643→              ))}
   644→            </ul>
   645→          )}
   646→        </>
   647→      )}
   648→    </Shell>
   649→  )
   650→}

   652→/* ================= Keyword ================= */
   653→
   654→function NeonKeyword({
   655→  site,
   656→  data,
   657→  loading,
   658→  onNavigate,
   659→  keyword,
   660→}: ThemeProps & { keyword: string }) {
   661→  const books = data.keywordBooks ?? []
   662→  const goMain = () =>
   663→    onNavigate(site.mainBookId ? { type: 'book', bookId: site.mainBookId } : { type: 'home' })
   664→  const related = Array.from(
   665→    new Set(
   666→      books
   667→        .flatMap((b) => [...b.keywords, ...b.suggestKeywords])
   668→        .filter((k) => k !== keyword),
   669→    ),
   670→  ).slice(0, 12)
   671→
   672→  return (
   673→    <Shell site={site} onNavigate={onNavigate}>
   674→      <Crumbs
   675→        trail={[
   676→          { label: '首页', onSelect: () => onNavigate({ type: 'home' }) },
   677→          { label: `关键词：${keyword}` },
   678→        ]}
   679→      />
   680→
   681→      {/* 最醒目的主书籍入口：渐变描边大按钮 */}
   682→      <div className={`mt-4 rounded-2xl ${GRAD_BR} p-[2px] shadow-[0_0_35px_rgba(192,38,211,0.3)]`}>
   683→        <button
   684→          type="button"
   685→          onClick={goMain}
   686→          className="group flex min-h-[44px] w-full items-center justify-between gap-4 rounded-[14px] bg-[#1d1d22] px-6 py-5 text-left"
   687→        >
   688→          <span className="flex items-center gap-3 text-base font-black text-white">
   689→            <Sparkles className="h-5 w-5 text-fuchsia-400" aria-hidden="true" />
   690→            进入主书籍信息页
   691→          </span>
   692→          <ArrowRight
   693→            className="h-5 w-5 text-fuchsia-400 transition-transform group-hover:translate-x-1"
   694→            aria-hidden="true"
   695→          />
   696→        </button>
   697→      </div>
   698→
   699→      <section className="py-10">
   700→        <p className="font-mono text-xs font-bold uppercase tracking-[0.4em] text-zinc-500">
   701→          Keyword
   702→        </p>
   703→        <h1 className="mt-4 text-4xl font-black tracking-tight sm:text-5xl">
   704→          <span className={GRAD_TEXT}>{keyword}</span>
   705→        </h1>
   706→        <p className="mt-4 text-sm text-zinc-400">与《{keyword}》相关的书籍与内容</p>
   707→      </section>
   708→
   709→      {loading ? (
   710→        <GridSkeleton count={3} />
   711→      ) : books.length === 0 ? (
   712→        <p className="py-20 text-center text-sm text-zinc-500">暂无相关书籍。</p>
   713→      ) : (
   714→        <section
   715→          aria-label="相关书籍"
   716→          className="grid grid-cols-1 gap-5 pb-4 sm:grid-cols-2 lg:grid-cols-3"
   717→        >
   718→          {books.map((book) => (
   719→            <NeonCard key={book.id} book={book} onNavigate={onNavigate} />
   720→          ))}
   721→        </section>
   722→      )}
   723→
   724→      {related.length > 0 && (
   725→        <section aria-label="相关关键词" className="mt-6 pb-4">
   726→          <h2 className="text-sm font-bold text-zinc-500">相关关键词</h2>
   727→          <div className="mt-3 flex flex-wrap gap-2">
   728→            {related.map((k) => (
   729→              <button
   730→                key={k}
   731→                type="button"
   732→                onClick={() => onNavigate({ type: 'keyword', keyword: k })}
   733→                className="min-h-[44px] rounded-full bg-white/5 px-4 text-sm font-semibold text-zinc-300 ring-1 ring-white/10 transition-all hover:text-fuchsia-300 hover:ring-fuchsia-400/50"
   734→              >
   735→                # {k}
   736→              </button>
   737→            ))}
   738→          </div>
   739→        </section>
   740→      )}
   741→    </Shell>
   742→  )
   743→}

   745→/* ================= 主组件 ================= */
   746→
   747→export function ThemeNeon(props: ThemeProps): ReactElement {
   748→  const { view } = props
   749→  switch (view.type) {
   750→    case 'home':
   751→      return <NeonHome {...props} />
   752→    case 'book':
   753→      return <NeonBook {...props} />
   754→    case 'toc':
   755→      return <NeonToc {...props} />
   756→    case 'chapter':
   757→      return <NeonChapter {...props} />
   758→    case 'keyword':
   759→      return <NeonKeyword {...props} keyword={view.keyword} />
   760→    default:
   761→      return <NeonHome {...props} />
   762→  }
   763→}
   764→
